# Isaac Sim Validation in Simulation Environment

## Overview

Validating Isaac Sim examples in the simulation environment is a critical step to ensure that all components work correctly before using them for educational purposes. This validation process confirms that scenes render properly, sensors function as expected, physics behave correctly, and synthetic datasets meet quality standards.

## Learning Objectives

By the end of this section, students will be able to:
- Set up and configure Isaac Sim for validation testing
- Validate basic and complex scenes with appropriate performance targets
- Test sensor functionality and data quality
- Verify physics simulation accuracy
- Assess synthetic dataset quality and completeness
- Troubleshoot common validation issues

## Validation Environment Setup

### Pre-Validation Requirements

Before running validation tests, ensure the following requirements are met:

        if not (self.isaac_sim_path / "isaac-sim.py").exists():
            print("❌ Isaac Sim executable not found")
            return False

        # Check if Isaac Sim can start
        try:
            # Test Isaac Sim startup without GUI
            result = subprocess.run([
                sys.executable, str(self.isaac_sim_path / "isaac-sim.py"),
                "--no-window", "--test-mode"
            ], timeout=30, capture_output=True, text=True)

            if result.returncode != 0:
                print(f"❌ Isaac Sim startup failed: {result.stderr}")
                return False

            print("✅ Isaac Sim installation validated")
            return True

        except subprocess.TimeoutExpired:
            print("⚠️  Isaac Sim startup timed out (may be normal)")
            return True  # Timeout might be normal in some environments
        except Exception as e:
            print(f"❌ Isaac Sim validation error: {str(e)}")
            return False

    def _test_gravity_simulation(self) -> dict:
        """Test gravity simulation accuracy"""

        print("   🌍 Testing gravity simulation...")

        # Simulate an object falling under gravity
        import numpy as np

        # Physics constants
        gravity = 9.81  # m/s^2
        initial_height = 10.0  # meters
        time_step = 0.01  # seconds
        simulation_time = 2.0  # seconds

        # Simulate falling motion
        time_points = np.arange(0, simulation_time, time_step)
        heights = initial_height - 0.5 * gravity * time_points**2

        # Check that object falls (height decreases)
        height_decreased = heights[-1] < heights[0]

        # Check that object hits ground (height approaches 0)
        hit_ground = heights[-1] <= 0

        result = {
            'test': 'gravity',
            'initial_height': initial_height,
            'final_height': float(heights[-1]),
            'time_elapsed': float(time_points[-1]),
            'height_decreased': height_decreased,
            'hit_ground': hit_ground,
            'passed': height_decreased
        }

        if result['passed']:
            print("   ✅ Gravity simulation test passed")
        else:
            print("   ❌ Gravity simulation test failed")

        return result

    def _test_collision_detection(self) -> dict:
        """Test collision detection"""

        print("   🔴 Testing collision detection...")

        # Simulate collision between two objects
        import numpy as np

        # Define two spheres that should collide
        sphere1_pos = np.array([0, 0, 0])
        sphere1_radius = 1.0

        sphere2_pos = np.array([1.5, 0, 0])  # Initially not colliding
        sphere2_radius = 1.0

        # Move sphere2 toward sphere1
        sphere2_pos = np.array([0.8, 0, 0])  # Now colliding

        # Calculate distance between centers
        distance = np.linalg.norm(sphere1_pos - sphere2_pos)
        radii_sum = sphere1_radius + sphere2_radius
        collision_detected = distance < radii_sum

        result = {
            'test': 'collision_detection',
            'sphere1_position': sphere1_pos.tolist(),
            'sphere2_position': sphere2_pos.tolist(),
            'sphere1_radius': sphere1_radius,
            'sphere2_radius': sphere2_radius,
            'distance': float(distance),
            'radii_sum': radii_sum,
            'collision_detected': collision_detected,
            'passed': collision_detected
        }

        if result['passed']:
            print("   ✅ Collision detection test passed")
        else:
            print("   ❌ Collision detection test failed")

        return result

        time_points = np.arange(0, duration, time_step)

            'acceleration': acceleration.tolist(),

    def _test_dataset_completeness(self) -> dict:
        """Test overall dataset completeness"""

        import os

        # Check that all modalities have the same number of samples
        rgb_dir = os.path.join(self.temp_dir, "rgb")
        depth_dir = os.path.join(self.temp_dir, "depth")
        semantic_dir = os.path.join(self.temp_dir, "semantic")

        rgb_count = len([f for f in os.listdir(rgb_dir) if f.endswith('.png')])
        depth_count = len([f for f in os.listdir(depth_dir) if f.endswith('.png')])
        semantic_count = len([f for f in os.listdir(semantic_dir) if f.endswith('.png')])

        # All counts should be equal and above target completeness
        all_equal = rgb_count == depth_count == semantic_count
        completeness = min(rgb_count, depth_count, semantic_count) / 10  # 10 was our target

        result = {
            'rgb_count': rgb_count,
            'depth_count': depth_count,
            'semantic_count': semantic_count,
            'all_modalities_equal': all_equal,
            'completeness_ratio': completeness,
            'passed': all_equal and completeness >= self.performance_targets['dataset_completeness']
        }

        if result['passed']:

        else:
            print(f"   ❌ Dataset completeness test failed (RGB:{rgb_count}, Depth:{depth_count}, Semantic:{semantic_count})")

        return result

                    results['installation'],
                    results['basic_scene_performance']['passed'],
                    results['complex_scene_performance']['passed'],
                    all(s['passed'] for s in results['sensor_functionality'].values()),
                    results['physics_simulation']['overall_passed'],
                    results['dataset_generation']['overall_passed']

        print(f"\n📋 Validation Summary:")
