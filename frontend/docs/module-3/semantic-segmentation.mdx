# Semantic Segmentation Mask Generation

## Overview

Semantic segmentation mask generation is a powerful capability in Isaac Sim that assigns a semantic label to each pixel in an image, indicating which object class it belongs to. This ground truth annotation is essential for training semantic segmentation models and is a key component of comprehensive synthetic datasets for computer vision applications.

- Optimize semantic segmentation generation for performance

## Semantic Segmentation Fundamentals

### Concept and Purpose

Semantic segmentation creates pixel-level annotations where each pixel is assigned to a semantic class:
- **Class-level labeling**: All pixels of the same class get the same label
- **Instance-agnostic**: Does not distinguish between different instances of the same class
- **Ground truth data**: Provides training labels for segmentation models

### Label Representation

Semantic segmentation masks are typically represented as:
- **Indexed images**: Each unique value represents a class
- **Color maps**: Each class assigned a specific color for visualization
- **JSON metadata**: Maps class IDs to class names

```python
def assign_semantic_label(prim_path, class_name, instance_id=None):
    """Assign semantic label to a prim in Isaac Sim"""

    stage = omni.usd.get_context().get_stage()
    prim = stage.GetPrimAtPath(prim_path)

        print(f"❌ Prim {prim_path} not found")

    semantic_api = replicator_omni.add_semantics(prim, class_name)

    if instance_id:
        # Set instance ID if needed
        semantic_api.set_instance_id(instance_id)

    return True
```

```python
def setup_semantic_segmentation():

    )
```

## Semantic Segmentation Quality and Validation

### Semantic Mask Validation

Validate the quality and accuracy of generated semantic masks:
```python
import numpy as np
import cv2
from collections import Counter

def validate_semantic_mask(semantic_path, expected_classes=None):
    """Validate semantic segmentation mask quality"""

    # Load semantic mask
    semantic_mask = cv2.imread(semantic_path, cv2.IMREAD_UNCHANGED)

    if semantic_mask is None:
        return False, "Invalid semantic mask file"

    # Get unique class IDs in the mask
    unique_classes = np.unique(semantic_mask)

    # Count pixels per class
    class_counts = Counter(semantic_mask.flatten())

    # Remove background (usually ID 0)
    if 0 in class_counts:
        del class_counts[0]

    validation_result = {
        'unique_classes': unique_classes,
        'class_pixel_counts': dict(class_counts),
        'total_pixels': semantic_mask.size,
        'valid_pixels': np.count_nonzero(semantic_mask),
        'valid_percentage': (np.count_nonzero(semantic_mask) / semantic_mask.size) * 100
    }

    # Check if expected classes are present
    if expected_classes:
        missing_classes = set(expected_classes) - set(unique_classes)
        if missing_classes:
            validation_result['missing_classes'] = list(missing_classes)
            return False, f"Missing expected classes: {missing_classes}"

    return True, validation_result

def analyze_semantic_distribution(semantic_path):
    """Analyze the distribution of semantic classes in a mask"""

    semantic_mask = cv2.imread(semantic_path, cv2.IMREAD_UNCHANGED)
    if semantic_mask is None:
        return None

    # Get unique values and their counts
    unique, counts = np.unique(semantic_mask, return_counts=True)

    # Calculate percentages
    total_pixels = semantic_mask.size
    percentages = (counts / total_pixels) * 100

    distribution = {}
    for i, class_id in enumerate(unique):
        distribution[int(class_id)] = {
            'count': int(counts[i]),
            'percentage': float(percentages[i])
        }

    return distribution
```

### Class Mapping and Visualization

Create visualizations and class mappings for semantic segmentation:

```python
import matplotlib.pyplot as plt
import matplotlib.patches as patches
from matplotlib.colors import ListedColormap
import random

def create_class_color_map(class_names):
    """Create a color map for semantic classes"""

    # Generate random colors for each class
    colors = {}
    for i, class_name in enumerate(class_names):
        # Generate a random color (R, G, B values between 0 and 1)
        color = (random.random(), random.random(), random.random())
        colors[i] = color

def visualize_semantic_mask(semantic_path, class_names, rgb_path=None):
    """Visualize semantic segmentation mask with class labels"""

    # Load semantic mask
    semantic_mask = cv2.imread(semantic_path, cv2.IMREAD_UNCHANGED)

    # Create color map
    class_ids = list(range(len(class_names) + 1))  # +1 for background
    class_names_with_bg = ['background'] + class_names

    # Generate random colors for visualization
    colors = []
    for i in range(len(class_ids)):
        if i == 0:  # Background
            colors.append([0, 0, 0])  # Black for background
        else:
            # Generate random color
            colors.append([random.random(), random.random(), random.random()])

    # Normalize colors to 0-1 range
    colors = np.array(colors)
    if np.max(colors) > 1:
        colors = colors / 255.0

    cmap = ListedColormap(colors)

    # Create visualization
    fig, axes = plt.subplots(1, 3 if rgb_path else 2, figsize=(15, 5))

    # Semantic mask
    axes[0].imshow(semantic_mask, cmap=cmap, vmin=0, vmax=len(class_ids)-1)
    axes[0].set_title('Semantic Segmentation')
    axes[0].axis('off')

    # Legend
    legend_elements = [patches.Patch(color=colors[i], label=class_names_with_bg[i])
                      for i in range(len(class_names_with_bg))]
    axes[1].legend(handles=legend_elements, loc='center')
    axes[1].axis('off')
    axes[1].set_title('Class Legend')

    # RGB image if provided
    if rgb_path:
        rgb_img = cv2.imread(rgb_path)
        rgb_img = cv2.cvtColor(rgb_img, cv2.COLOR_BGR2RGB)
        axes[2].imshow(rgb_img)
        axes[2].set_title('RGB Image')
        axes[2].axis('off')

    plt.tight_layout()
    plt.show()
```

## Troubleshooting Common Semantic Issues

### Label Assignment Issues
- **Missing labels**: Verify that semantic schema is properly applied to objects
- **Incorrect labels**: Check that class names match expected values
- **No semantic output**: Ensure annotator is attached to render product

### Quality Issues
- **Inconsistent boundaries**: Check object overlaps and geometry
- **Missing regions**: Verify objects are within camera view and clipping range
- **Wrong class IDs**: Check semantic schema assignment and naming

### Performance Issues
- **Slow generation**: Semantic segmentation is typically fast, but complex scenes may slow it down
- **Memory issues**: Monitor memory usage with multiple sensors

### Semantic + Instance Segmentation
Combine semantic and instance segmentation for comprehensive annotations:
```python
def process_semantic_instance_masks(semantic_path, instance_path):
    """Process combined semantic and instance segmentation"""

    # Load semantic mask
    semantic_mask = cv2.imread(semantic_path, cv2.IMREAD_UNCHANGED)

    # Load instance mask
    instance_mask = cv2.imread(instance_path, cv2.IMREAD_UNCHANGED)

    # Combine semantic and instance information
    # This creates a mapping from instance ID to semantic class
    instance_to_semantic = {}

    for instance_id in np.unique(instance_mask):
        if instance_id == 0:  # Skip background
            continue

        # Find the semantic class for this instance
        instance_pixels = (instance_mask == instance_id)
        semantic_values = semantic_mask[instance_pixels]

        # Most common semantic value for this instance
        unique, counts = np.unique(semantic_values, return_counts=True)
        semantic_class = unique[np.argmax(counts)]

        instance_to_semantic[int(instance_id)] = int(semantic_class)

    return instance_to_semantic
```
