        self.max_sensors = max_sensors
        self.sensors: Dict[str, SensorConfig] = {}
        self.active_sensors: List[str] = []

        """Add a sensor to the simulation"""

        if len(self.sensors) >= self.max_sensors:
            return False

        self.sensors[config.sensor_id] = config
        self.sensor_data_buffers[config.sensor_id] = asyncio.Queue(maxsize=10)  # Buffer size
        return True

    def remove_sensor(self, sensor_id: str) -> bool:
        """Remove a sensor from the simulation"""

        if sensor_id in self.sensors:
            del self.sensors[sensor_id]
            if sensor_id in self.sensor_data_buffers:
                del self.sensor_data_buffers[sensor_id]
            return True
        return False

    def get_sensor_by_type(self, sensor_type: SensorType) -> List[str]:
        """Get all sensors of a specific type"""

        return [sid for sid, config in self.sensors.items()
                if config.sensor_type == sensor_type]

    def get_sensors_by_priority(self, priority: SensorPriority) -> List[str]:
        """Get all sensors of a specific priority level"""

        return [sid for sid, config in self.sensors.items()

import time
from collections import defaultdict, deque

class SensorUpdateScheduler:
    """Schedule sensor updates to optimize performance with 50+ sensors"""

    def __init__(self):
        self.update_schedule = defaultdict(deque)  # Priority -> queue of sensor IDs
        self.last_update_time = {}  # sensor_id -> last_update_time
        self.update_intervals = {}  # sensor_id -> update_interval (seconds)
        self.frame_counter = 0
        self.frame_rate = 60  # Base simulation frame rate

    def register_sensor(self, sensor_id: str, config: SensorConfig):
        """Register a sensor with the scheduler"""

        # Calculate update interval based on desired rate
        update_interval = 1.0 / config.update_rate if config.update_rate > 0 else 0.001
        self.update_intervals[sensor_id] = update_interval
        self.last_update_time[sensor_id] = 0

        # Add to appropriate priority queue
        self.update_schedule[config.priority.value].append(sensor_id)

    def get_sensors_to_update(self, current_time: float) -> List[str]:
        """Get list of sensors that need updating at current time"""

        sensors_to_update = []

        # Check each priority level
        for priority_level in sorted(self.update_schedule.keys()):
            queue = self.update_schedule[priority_level]
            temp_queue = deque()

            while queue:
                sensor_id = queue.popleft()
                last_update = self.last_update_time.get(sensor_id, 0)
                interval = self.update_intervals[sensor_id]

                if current_time - last_update >= interval:
                    sensors_to_update.append(sensor_id)
                    self.last_update_time[sensor_id] = current_time
                else:
                    temp_queue.append(sensor_id)

            # Put remaining sensors back in the queue
            self.update_schedule[priority_level] = temp_queue

        # Limit updates per frame to prevent performance issues
        max_updates_per_frame = 10  # Adjust based on performance requirements
        return sensors_to_update[:max_updates_per_frame]

    def update_frame_counter(self):
        """Update frame counter for frame-based scheduling"""
        self.frame_counter += 1

class SensorResourceLimiter:
    """Limit sensor resource usage to maintain performance"""

    def __init__(self):
        self.max_bandwidth_per_frame = 500  # Mbps
        self.current_bandwidth_usage = 0.0
        self.max_compute_units = 80  # Percentage of GPU compute units
        self.current_compute_usage = 0.0

    def can_schedule_sensor_update(self, sensor_config: SensorConfig) -> bool:
        """Check if a sensor update can be scheduled based on resource limits"""

        # Estimate resource requirements
        estimated_bandwidth = sensor_config.bandwidth_limit or self._estimate_bandwidth(sensor_config)
        estimated_compute = self._estimate_compute_usage(sensor_config)

        # Check if within limits
        can_schedule = (
            self.current_bandwidth_usage + estimated_bandwidth <= self.max_bandwidth_per_frame and
            self.current_compute_usage + estimated_compute <= self.max_compute_units
        )

        return can_schedule

    def _estimate_bandwidth(self, config: SensorConfig) -> float:
        """Estimate bandwidth usage for a sensor"""
        if config.sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA]:
            # Estimate based on resolution and frame rate
            pixels_per_frame = config.resolution[0] * config.resolution[1]
            bytes_per_pixel = 3 if config.sensor_type == SensorType.RGB_CAMERA else 4
            estimated_mbps = (pixels_per_frame * bytes_per_pixel * 8 * config.update_rate) / (1024 * 1024)
            return estimated_mbps
        else:
            return 1.0  # Low bandwidth for other sensor types

    def _estimate_compute_usage(self, config: SensorConfig) -> float:
        """Estimate compute unit usage for a sensor"""
        if config.sensor_type in [SensorType.LIDAR]:
            return 15.0  # Higher compute for LiDAR
        elif config.sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA]:
            return 10.0  # Medium compute for cameras
        else:
            return 5.0  # Low compute for other sensors

    def reserve_resources(self, sensor_config: SensorConfig):
        """Reserve resources for a sensor update"""
        estimated_bandwidth = sensor_config.bandwidth_limit or self._estimate_bandwidth(sensor_config)
        estimated_compute = self._estimate_compute_usage(sensor_config)

        self.current_bandwidth_usage += estimated_bandwidth
        self.current_compute_usage += estimated_compute

    def release_resources(self, sensor_config: SensorConfig):
        """Release resources after sensor update"""
        estimated_bandwidth = sensor_config.bandwidth_limit or self._estimate_bandwidth(sensor_config)
        estimated_compute = self._estimate_compute_usage(sensor_config)

        self.current_bandwidth_usage = max(0, self.current_bandwidth_usage - estimated_bandwidth)
        self.current_compute_usage = max(0, self.current_compute_usage - estimated_compute)
```

class SharedRenderingPipeline:
    """Optimize rendering for multiple sensors by sharing calculations"""

    def __init__(self):
        self.shared_geometries = {}  # Cache for shared geometry calculations
        self.shared_lighting = {}    # Cache for shared lighting calculations
        self.render_targets = {}     # Multi-render target setup
        self.occlusion_cache = {}    # Cache for occlusion calculations

    def setup_multi_sensor_rendering(self, sensors: List[SensorConfig]) -> dict:
        """Set up shared rendering for multiple sensors"""

        # Group sensors by similar properties to maximize sharing
        camera_groups = self._group_similar_cameras(sensors)
        lidar_groups = self._group_similar_lidars(sensors)

        setup_config = {
            'camera_groups': camera_groups,
            'lidar_groups': lidar_groups,
            'shared_render_targets': self._setup_shared_render_targets(camera_groups),
            'geometry_cache_size': self._calculate_geometry_cache_size(sensors),
            'lighting_cache_size': self._calculate_lighting_cache_size(sensors)
        }

        return setup_config

    def _group_similar_cameras(self, sensors: List[SensorConfig]) -> List[List[SensorConfig]]:
        """Group cameras with similar properties for shared rendering"""

        camera_sensors = [s for s in sensors if s.sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA]]
        groups = []

        # Group by resolution
        resolution_groups = {}
        for sensor in camera_sensors:
            res_key = sensor.resolution
            if res_key not in resolution_groups:
                resolution_groups[res_key] = []
            resolution_groups[res_key].append(sensor)

        # Further group by field of view
        for res_key, res_sensors in resolution_groups.items():
            fov_groups = {}
            for sensor in res_sensors:
                fov_key = round(sensor.fov, 1)  # Round to nearest 0.1 degree
                if fov_key not in fov_groups:
                    fov_groups[fov_key] = []
                fov_groups[fov_key].append(sensor)

            for fov_key, fov_sensors in fov_groups.items():
                groups.append(fov_sensors)

        return groups

    def _group_similar_lidars(self, sensors: List[SensorConfig]) -> List[List[SensorConfig]]:
        """Group LiDAR sensors with similar properties"""

        lidar_sensors = [s for s in sensors if s.sensor_type == SensorType.LIDAR]
        groups = []

        # Group by range and resolution
        range_groups = {}
        for sensor in lidar_sensors:
            range_key = (sensor.range_min, sensor.range_max)
            if range_key not in range_groups:
                range_groups[range_key] = []
            range_groups[range_key].append(sensor)

        for range_key, range_sensors in range_groups.items():
            groups.append(range_sensors)

        return groups

    def _setup_shared_render_targets(self, camera_groups: List[List[SensorConfig]]) -> dict:
        """Set up shared render targets for camera groups"""

        render_targets = {}
        for i, group in enumerate(camera_groups):
            if group:
                # Use the resolution of the first sensor in the group
                resolution = group[0].resolution
                render_target_id = f"shared_rt_{i}_{resolution[0]}x{resolution[1]}"

                render_targets[render_target_id] = {
                    'resolution': resolution,
                    'sensors': [s.sensor_id for s in group],
                    'format': 'RGBA8' if group[0].sensor_type == SensorType.RGB_CAMERA else 'R32F'
                }

        return render_targets

    def _calculate_geometry_cache_size(self, sensors: List[SensorConfig]) -> int:
        """Calculate appropriate geometry cache size based on sensor count"""

        # More sensors = more cache needed for frequently accessed geometry
        base_cache_size = 100 * 1024 * 1024  # 100MB base
        per_sensor_increase = 2 * 1024 * 1024  # 2MB per sensor

        return base_cache_size + (len(sensors) * per_sensor_increase)

    def _calculate_lighting_cache_size(self, sensors: List[SensorConfig]) -> int:
        """Calculate appropriate lighting cache size"""

        # Lighting cache scales with scene complexity, not just sensor count
        return 50 * 1024 * 1024  # 50MB for lighting cache
```

        reduction_factor = 1.0 - target_reduction

        # Ensure minimum resolution
        new_width = max(320, new_width)  # Minimum 320 pixels width
        new_height = max(240, new_height)  # Minimum 240 pixels height

class GPUResourceManager:
    """Manage GPU resources for high-density sensor deployment"""

    def __init__(self):
        self.max_gpu_memory_per_frame = 1024 * 1024 * 1024  # 1GB per frame
        self.max_compute_units = 80  # Percentage of GPU compute units
        self.current_gpu_memory_usage = 0
        self.current_compute_usage = 0
        self.frame_gpu_memory_usage = 0
        self.sensor_gpu_allocations = {}  # sensor_id -> GPU memory allocation

    def allocate_gpu_resources(self, sensor_id: str, config: SensorConfig) -> bool:
        """Allocate GPU resources for a sensor"""

        required_memory = self._calculate_gpu_memory_requirement(config)
        required_compute = self._calculate_compute_requirement(config)

        # Check if resources are available within limits
        if (self.current_gpu_memory_usage + required_memory <= self.max_gpu_memory_per_frame and
            self.current_compute_usage + required_compute <= self.max_compute_units and
            self.frame_gpu_memory_usage + required_memory <= self.max_gpu_memory_per_frame):

            self.current_gpu_memory_usage += required_memory
            self.current_compute_usage += required_compute
            self.frame_gpu_memory_usage += required_memory
            self.sensor_gpu_allocations[sensor_id] = required_memory

            return True

        return False

    def _calculate_gpu_memory_requirement(self, config: SensorConfig) -> int:
        """Calculate GPU memory requirement for a sensor"""

        if config.sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA]:
            # Calculate based on resolution and data type
            pixels = config.resolution[0] * config.resolution[1]
            bytes_per_pixel = 4 if config.sensor_type == SensorType.RGB_CAMERA else 8  # Depth needs more precision
            # Add buffer for processing
            return pixels * bytes_per_pixel * 2
        elif config.sensor_type == SensorType.LIDAR:
            # LiDAR needs memory for point cloud processing
            points = int(config.range_max * 1000)  # Estimate number of points
            return points * 32  # 32 bytes per point (position + intensity + other data)
        else:
            # Other sensors need minimal GPU memory
            return 1024 * 1024  # 1MB

    def _calculate_compute_requirement(self, config: SensorConfig) -> float:
        """Calculate compute unit requirement for a sensor"""

        if config.sensor_type == SensorType.LIDAR:
            return 20.0  # High compute for ray tracing
        elif config.sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA]:
            return 10.0  # Medium compute for rendering
        else:
            return 2.0  # Low compute for other sensors

    def release_gpu_resources(self, sensor_id: str):
        """Release GPU resources for a sensor"""

        if sensor_id in self.sensor_gpu_allocations:
            released_memory = self.sensor_gpu_allocations[sensor_id]
            self.current_gpu_memory_usage = max(0, self.current_gpu_memory_usage - released_memory)
            # Compute usage would need to be tracked separately per sensor

    def reset_frame_counters(self):
        """Reset frame-based counters at the start of each frame"""
        self.frame_gpu_memory_usage = 0

    def get_gpu_utilization_report(self) -> dict:
        """Get current GPU utilization report"""

        return {
            'current_memory_usage': self.current_gpu_memory_usage,
            'max_memory_per_frame': self.max_gpu_memory_per_frame,
            'memory_utilization': self.current_gpu_memory_usage / self.max_gpu_memory_per_frame,
            'current_compute_usage': self.current_compute_usage,
            'max_compute_units': self.max_compute_units,
            'compute_utilization': self.current_compute_usage / self.max_compute_units,
            'sensors_managed': len(self.sensor_gpu_allocations)
        }

            'bandwidth_efficiency': 0.1
        }

        overall_quality = sum(
            quality_metrics[metric] * weights[metric]
            for metric in quality_metrics
        )

        quality_report = {
            'sensor_id': sensor_id,
            'timestamp': time.time(),
            'quality_metrics': quality_metrics,
            'overall_quality': overall_quality,
            'quality_status': 'PASS' if overall_quality >= 0.8 else 'WARN' if overall_quality >= 0.6 else 'FAIL',
            'recommendations': self._generate_recommendations(quality_metrics, config)
        }

        # Store in history
        if sensor_id not in self.quality_history:
            self.quality_history[sensor_id] = []
        self.quality_history[sensor_id].append(overall_quality)

        return quality_report

    def _measure_latency(self, sensor_id: str) -> float:
        """Measure latency for a sensor (1.0 = no latency, 0.0 = max latency)"""

        # This would measure actual latency
        # For simulation, return a value based on system load
        import random
        return random.uniform(0.7, 1.0)  # Simulated good latency

    def _measure_completeness(self, sensor_id: str, data: Any) -> float:
        """Measure data completeness (1.0 = complete, 0.0 = missing)"""

        if data is None:
            return 0.0

        # For different data types, measure completeness differently
        if isinstance(data, dict) and 'data' in data:
            expected_size = getattr(self, f'_get_expected_size_for_{sensor_id}', lambda: 100)()
            actual_size = len(data.get('data', []))
            return min(1.0, actual_size / expected_size) if expected_size > 0 else 0.0
        else:
            return 1.0  # Assume complete if not a dictionary with data

    def _measure_accuracy(self, sensor_id: str, data: Any, config: SensorConfig) -> float:
        """Measure data accuracy (1.0 = perfectly accurate, 0.0 = completely inaccurate)"""

        # Accuracy would depend on sensor type and noise models
        # For simulation, return based on sensor priority
        if config.priority in [SensorPriority.CRITICAL, SensorPriority.HIGH]:
            return 0.98
        elif config.priority == SensorPriority.MEDIUM:
            return 0.95
        else:
            return 0.90

    def _measure_bandwidth_efficiency(self, sensor_id: str) -> float:
        """Measure bandwidth efficiency (1.0 = perfectly efficient, 0.0 = inefficient)"""

        # This would measure actual vs allocated bandwidth usage
        return 0.85  # Simulated efficiency

    def _generate_recommendations(self, quality_metrics: dict, config: SensorConfig) -> List[str]:
        """Generate recommendations based on quality metrics"""

        recommendations = []

        if quality_metrics['latency'] < 0.8:
            recommendations.append("Consider reducing update rate for better latency")
        if quality_metrics['data_completeness'] < 0.9:
            recommendations.append("Check for buffer overflows or network issues")
        if quality_metrics['accuracy'] < 0.95 and config.priority in [SensorPriority.CRITICAL, SensorPriority.HIGH]:
            recommendations.append("Verify sensor calibration and noise parameters")

        return recommendations

    def get_system_quality_overview(self) -> dict:
        """Get overall system quality metrics"""

        if not self.quality_history:
            return {"status": "no_data", "message": "No quality data collected"}

        # Calculate average quality across all sensors
        all_quality_scores = []
        for sensor_scores in self.quality_history.values():
            if sensor_scores:
                all_quality_scores.append(sensor_scores[-1])  # Last score per sensor

        if not all_quality_scores:
            return {"status": "no_recent_data", "message": "No recent quality data"}

        avg_quality = statistics.mean(all_quality_scores)
        min_quality = min(all_quality_scores)
        max_quality = max(all_quality_scores)

        overview = {
            "status": "active",
            "total_sensors_monitored": len(self.quality_history),
            "average_quality": avg_quality,
            "min_quality": min_quality,
            "max_quality": max_quality,
            "quality_stability": statistics.stdev(all_quality_scores) if len(all_quality_scores) > 1 else 0,
            "sensors_above_threshold": sum(1 for q in all_quality_scores if q >= 0.8),
            "sensors_below_threshold": sum(1 for q in all_quality_scores if q < 0.6)
        }

        return overview

def plan_sensor_deployment(num_sensors: int) -> List[SensorConfig]:
    """Plan deployment of various sensor types"""

    configs = []

    # Distribute sensors across types and priorities
    sensors_per_type = num_sensors // 7  # Roughly equal distribution
    remainder = num_sensors % 7

    sensor_counts = {
        SensorType.RGB_CAMERA: sensors_per_type + (1 if remainder > 0 else 0),
        SensorType.DEPTH_CAMERA: sensors_per_type + (1 if remainder > 1 else 0),
        SensorType.LIDAR: sensors_per_type + (1 if remainder > 2 else 0),
        SensorType.IMU: sensors_per_type + (1 if remainder > 3 else 0),
        SensorType.GPS: sensors_per_type + (1 if remainder > 4 else 0),
        SensorType.CONTACT_SENSOR: sensors_per_type + (1 if remainder > 5 else 0),
        SensorType.FORCE_TORQUE: sensors_per_type + (1 if remainder > 6 else 0)
    }

    sensor_id_counter = 0

    for sensor_type, count in sensor_counts.items():
        for i in range(count):
            # Determine priority based on sensor type
            if sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA, SensorType.LIDAR]:
                priority = SensorPriority.HIGH if i < count // 3 else SensorPriority.MEDIUM
            else:
                priority = SensorPriority.MEDIUM if i < count // 2 else SensorPriority.LOW

            # Determine update rate based on type
            update_rates = {
                SensorType.RGB_CAMERA: 30,
                SensorType.DEPTH_CAMERA: 30,
                SensorType.LIDAR: 10,
                SensorType.IMU: 200,
                SensorType.GPS: 10,
                SensorType.CONTACT_SENSOR: 60,
                SensorType.FORCE_TORQUE: 500
            }

            config = SensorConfig(
                sensor_id=f"{sensor_type.value}_{sensor_id_counter:03d}",
                sensor_type=sensor_type,
                position=(i * 0.5, i * 0.5, 1.0),  # Distribute in space
                orientation=(0, 0, 0, 1),  # Default orientation
                priority=priority,
                update_rate=update_rates[sensor_type],
                resolution=(640, 480) if sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA] else (0, 0),
                fov=60.0 if sensor_type in [SensorType.RGB_CAMERA, SensorType.DEPTH_CAMERA] else 0.0,
                range_min=0.1 if sensor_type in [SensorType.LIDAR] else 0.0,
                range_max=25.0 if sensor_type in [SensorType.LIDAR] else 0.0,
                noise_model="gaussian" if sensor_type != SensorType.CONTACT_SENSOR else None
            )

            configs.append(config)
            sensor_id_counter += 1

    return configs

## Troubleshooting High-Density Sensor Issues

#### Resource Exhaustion
- **Symptoms**: System crashes, sensor failures, poor performance
- **Solutions**:
  - Implement resource limits and monitoring
  - Use dynamic resource allocation
  - Apply sensor prioritization
  - Reduce sensor update rates when needed

#### Data Quality Degradation
- **Symptoms**: Missing data, poor accuracy, high latency
- **Solutions**:
  - Implement quality monitoring systems
  - Use data compression and streaming
  - Apply sensor-specific optimizations
  - Validate data completeness regularly

#### Performance Bottlenecks
- **Symptoms**: Low FPS, high latency, dropped frames
- **Solutions**:
  - Optimize rendering pipelines
  - Use shared calculations where possible
  - Implement efficient scheduling
  - Monitor and manage resource usage
