# Isaac ROS VSLAM Pipeline Implementation

## Overview

The Isaac ROS VSLAM (Visual Simultaneous Localization and Mapping) pipeline connects Isaac Sim with ROS 2 for hardware-accelerated perception. This pipeline enables real-time visual SLAM processing using Isaac Sim's high-fidelity simulation and ROS 2's flexible robotics framework. The pipeline processes sensor data from Isaac Sim, performs VSLAM operations, and provides pose estimates that can be used for navigation and other robotic applications.

## Learning Objectives

By the end of this section, students will be able to:
- Understand the architecture of Isaac ROS VSLAM pipeline
- Configure Isaac ROS VSLAM nodes for processing Isaac Sim sensor data
- Integrate Isaac Sim outputs with Isaac ROS perception pipelines
- Validate VSLAM performance and accuracy
- Troubleshoot common VSLAM pipeline issues
- Optimize VSLAM processing for target frame rates (10-30 Hz)

## Isaac Sim to Isaac ROS Integration

### Sensor Bridge Configuration

Configure the bridge between Isaac Sim sensors and ROS topics:

```python
import omni
import omni.replicator.core as rep
import omni.isaac.core.utils.prims as prims_utils
from pxr import UsdGeom, Gf
import carb

class IsaacSimSensorBridge:
    """Bridge Isaac Sim sensors to ROS topics for VSLAM pipeline"""

    def __init__(self):
        self.stage = omni.usd.get_context().get_stage()
        self.camera = None
        self.rgb_annotator = None
        self.depth_annotator = None
        self.isaac_bridge = None

    def setup_sensor_bridge(self):
        """Set up the sensor bridge from Isaac Sim to ROS"""

        # Create camera in Isaac Sim
        self.camera = rep.create.camera()

        # Create render product for the camera
        render_product = rep.create.render_product(self.camera, (1920, 1080))

        # Attach RGB annotator
        self.rgb_annotator = rep.AnnotatorRegistry.get_annotator("rgb")
        self.rgb_annotator.attach(render_product)

        # Attach depth annotator
        self.depth_annotator = rep.AnnotatorRegistry.get_annotator("distance_to_image_plane")
        self.depth_annotator.attach(render_product)

        # Configure the bridge to publish to ROS topics
        self.configure_ros_bridge()

    def configure_ros_bridge(self):
        """Configure ROS bridge parameters"""

        # Set up ROS topic remappings
        self.ros_remappings = {
            # Camera image topics
            '/camera/color/image_raw': '/rgb/image_raw',
            '/camera/color/camera_info': '/rgb/camera_info',
            '/camera/depth/image_rect_raw': '/depth/image_rect_raw',

            # VSLAM topics
            '/visual_slam/tracking/pose': '/robot_pose',
            '/visual_slam/tracking/odometry': '/odometry',
            '/visual_slam/map': '/map',
            '/visual_slam/visual_odometry': '/visual_odometry',

            # TF topics
            '/tf': '/tf',
            '/tf_static': '/tf_static'
        }

        # Configure QoS settings for reliable communication
        self.qos_profiles = {
            'sensor_data': {
                'history': 'keep_last',
                'depth': 5,
                'reliability': 'reliable',
                'durability': 'volatile'
            },
            'default': {
                'history': 'keep_last',
                'depth': 10,
                'reliability': 'reliable',
                'durability': 'volatile'
            }
        }

    def start_data_streaming(self):
        """Start streaming sensor data from Isaac Sim to ROS"""

        # Define the execution graph for data streaming
        with rep.orchestrator.add_on_frame():
            # Get RGB data
            rgb_data = self.rgb_annotator.get_data()

            # Get depth data
            depth_data = self.depth_annotator.get_data()

            # Publish to ROS topics
            self.publish_to_ros('/rgb/image_raw', rgb_data)
            self.publish_to_ros('/depth/image_rect_raw', depth_data)

            # Process and publish pose estimates
            if hasattr(self, 'vslam_processor'):
                pose_estimate = self.vslam_processor.process_frame(rgb_data, depth_data)
                self.publish_to_ros('/robot_pose', pose_estimate)

    def publish_to_ros(self, topic: str, data):
        """Publish data to ROS topic"""

        # This would interface with the ROS bridge
        # In Isaac Sim, this is typically done using the ROS Bridge extension

        # Example implementation:
        if self.isaac_bridge:
            self.isaac_bridge.publish(topic, data)

    def setup_coordinate_system(self):
        """Set up coordinate system transformation"""

        # Isaac Sim uses Z-up coordinate system
        # ROS uses Z-forward coordinate system
        # Need to transform between coordinate systems

        self.coordinate_transform = {
            'isaac_to_ros': {
                'translation': [0, 0, 0],  # No translation change
                'rotation': [0.5, 0.5, -0.5, 0.5]  # Rotate -90 degrees around X axis
            },
            'ros_to_isaac': {
                'translation': [0, 0, 0],
                'rotation': [0.5, -0.5, 0.5, 0.5]  # Rotate 90 degrees around X axis
            }
        }

    def validate_sensor_connection(self) -> bool:
        """Validate that sensor data is flowing correctly"""

        try:
            # Check if camera is properly configured
            if not self.camera:
                carb.log_error("Camera not configured")
                return False

            # Check if annotators are attached
            if not self.rgb_annotator or not self.depth_annotator:
                carb.log_error("Annotators not properly attached")
                return False

            # Test data flow
            rgb_data = self.rgb_annotator.get_data()
            depth_data = self.depth_annotator.get_data()

            if rgb_data is None or depth_data is None:
                carb.log_error("No sensor data available")
                return False

            # Check data dimensions
            if 'data' in rgb_data and 'data' in depth_data:
                rgb_shape = rgb_data['data'].shape
                depth_shape = depth_data['data'].shape

                if rgb_shape[0:2] != depth_shape[0:2]:
                    carb.log_error(f"RGB and depth dimensions don't match: {rgb_shape} vs {depth_shape}")
                    return False

            carb.log_info("Sensor connection validated successfully")
            return True

        except Exception as e:
            carb.log_error(f"Sensor connection validation failed: {str(e)}")
            return False
```

### VSLAM Pipeline Implementation

Implement the complete VSLAM pipeline:

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, CameraInfo
from geometry_msgs.msg import PoseStamped, TransformStamped
from nav_msgs.msg import Odometry
from tf2_ros import TransformBroadcaster
import cv2
import numpy as np
from cv_bridge import CvBridge
from typing import Optional

class IsaacROSVisualSLAM(Node):
    """Isaac ROS Visual SLAM Node implementation"""

    def __init__(self):
        super().__init__('visual_slam_node')

        # Declare parameters
        self.declare_parameter('use_sim_time', True)
        self.declare_parameter('enable_mapping', True)
        self.declare_parameter('enable_loop_closure', True)
        self.declare_parameter('input_width', 1920)
        self.declare_parameter('input_height', 1080)
        self.declare_parameter('publish_odom_tf', True)
        self.declare_parameter('map_frame', 'map')
        self.declare_parameter('odom_frame', 'odom')
        self.declare_parameter('base_frame', 'base_link')

        # Get parameters
        self.use_sim_time = self.get_parameter('use_sim_time').value
        self.enable_mapping = self.get_parameter('enable_mapping').value
        self.enable_loop_closure = self.get_parameter('enable_loop_closure').value
        self.input_width = self.get_parameter('input_width').value
        self.input_height = self.get_parameter('input_height').value
        self.publish_odom_tf = self.get_parameter('publish_odom_tf').value
        self.map_frame = self.get_parameter('map_frame').value
        self.odom_frame = self.get_parameter('odom_frame').value
        self.base_frame = self.get_parameter('base_frame').value

        # Initialize CV bridge
        self.bridge = CvBridge()

        # Initialize pose and odometry publishers
        self.pose_publisher = self.create_publisher(PoseStamped, '/robot_pose', 10)
        self.odom_publisher = self.create_publisher(Odometry, '/odometry', 10)
        self.map_publisher = self.create_publisher(OccupancyGrid, '/map', 10) if self.enable_mapping else None

        # Initialize TF broadcaster
        if self.publish_odom_tf:
            self.tf_broadcaster = TransformBroadcaster(self)

        # Initialize image subscriber
        self.image_subscription = self.create_subscription(
            Image,
            '/rgb/image_raw',
            self.image_callback,
            10
        )

        # Initialize camera info subscriber
        self.camera_info_subscription = self.create_subscription(
            CameraInfo,
            '/rgb/camera_info',
            self.camera_info_callback,
            10
        )

        # VSLAM processing variables
        self.previous_image = None
        self.current_pose = np.eye(4)  # 4x4 identity matrix
        self.pose_history = []
        self.feature_tracker = None
        self.map_builder = None
        self.loop_detector = None

        # Initialize VSLAM components
        self.initialize_vslam_components()

        # Performance metrics
        self.frame_count = 0
        self.start_time = self.get_clock().now()

        self.get_logger().info('Isaac ROS Visual SLAM Node initialized')

    def initialize_vslam_components(self):
        """Initialize VSLAM processing components"""

        try:
            # Initialize feature detector and descriptor extractor
            self.feature_detector = cv2.ORB_create(
                nfeatures=1000,
                scaleFactor=1.2,
                nlevels=8,
                edgeThreshold=31,
                patchSize=31,
                fastThreshold=20
            )

            # Initialize feature matcher
            self.feature_matcher = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=False)

            # Initialize KLT tracker for optical flow
            self.lk_params = dict(
                winSize=(21, 21),
                maxLevel=3,
                criteria=(cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 30, 0.03)
            )

            # Initialize pose estimator
            self.ransac_reproj_threshold = 5.0

            self.get_logger().info('VSLAM components initialized successfully')

        except Exception as e:
            self.get_logger().error(f'Failed to initialize VSLAM components: {str(e)}')

    def image_callback(self, msg):
        """Process incoming image messages"""

        try:
            # Convert ROS image to OpenCV format
            cv_image = self.bridge.imgmsg_to_cv2(msg, desired_encoding='passthrough')

            # Convert to grayscale if needed
            if len(cv_image.shape) == 3:
                gray_image = cv2.cvtColor(cv_image, cv2.COLOR_RGB2GRAY)
            else:
                gray_image = cv_image

            # Process frame with VSLAM pipeline
            pose_estimate = self.process_vslam_frame(gray_image, msg.header.stamp)

            if pose_estimate is not None:
                # Publish pose estimate
                self.publish_pose_estimate(pose_estimate, msg.header.stamp)

                # Publish odometry
                self.publish_odometry(pose_estimate, msg.header.stamp)

                # Broadcast TF if enabled
                if self.publish_odom_tf:
                    self.broadcast_transforms(pose_estimate, msg.header.stamp)

                # Update performance metrics
                self.frame_count += 1
                current_time = self.get_clock().now()
                elapsed = (current_time - self.start_time).nanoseconds / 1e9
                if elapsed > 0:
                    fps = self.frame_count / elapsed
                    if self.frame_count % 100 == 0:  # Log every 100 frames
                        self.get_logger().info(f'VSLAM Processing FPS: {fps:.2f}')

        except Exception as e:
            self.get_logger().error(f'Error processing image: {str(e)}')

    def camera_info_callback(self, msg):
        """Process camera calibration information"""

        # Store camera intrinsics
        self.camera_matrix = np.array(msg.k).reshape(3, 3)
        self.distortion_coeffs = np.array(msg.d)

        # Set up projection matrix
        self.projection_matrix = np.hstack([self.camera_matrix, np.zeros((3, 1))])

    def process_vslam_frame(self, current_image, timestamp):
        """Process a single frame through the VSLAM pipeline"""

        if self.previous_image is None:
            # First frame - set as reference
            self.previous_image = current_image
            self.reference_features = self.extract_features(current_image)
            return self.current_pose

        # Extract features from current frame
        current_features = self.extract_features(current_image)

        if current_features is None or self.reference_features is None:
            return self.current_pose

        # Match features between frames
        matches = self.match_features(self.reference_features, current_features)

        if len(matches) < 10:  # Not enough matches for reliable pose estimation
            return self.current_pose

        # Extract matched points
        prev_pts, curr_pts = self.extract_matched_points(matches, self.reference_features, current_features)

        if len(prev_pts) < 10:
            return self.current_pose

        # Estimate motion using Essential Matrix
        try:
            E, mask = cv2.findEssentialMat(
                curr_pts, prev_pts,
                self.camera_matrix,
                method=cv2.RANSAC,
                prob=0.999,
                threshold=self.ransac_reproj_threshold
            )

            if E is not None:
                # Recover pose from Essential Matrix
                _, R, t, _ = cv2.recoverPose(E, curr_pts, prev_pts, self.camera_matrix)

                # Create transformation matrix
                transformation = np.eye(4)
                transformation[:3, :3] = R
                transformation[:3, 3] = t.flatten()

                # Update current pose
                self.current_pose = self.current_pose @ np.linalg.inv(transformation)

                # Update reference for next iteration
                self.previous_image = current_image
                self.reference_features = current_features

                return self.current_pose

        except Exception as e:
            self.get_logger().warn(f'Pose estimation failed: {str(e)}')

        return self.current_pose

    def extract_features(self, image):
        """Extract ORB features from image"""

        try:
            kp, des = self.feature_detector.detectAndCompute(image, None)
            if kp is not None and des is not None:
                return (kp, des)
        except Exception as e:
            self.get_logger().warn(f'Feature extraction failed: {str(e)}')

        return None

    def match_features(self, desc1, desc2):
        """Match features between two descriptors"""

        try:
            kp1, des1 = desc1
            kp2, des2 = desc2

            if des1 is None or des2 is None or len(des1) == 0 or len(des2) == 0:
                return []

            # Use FLANN matcher for better performance
            FLANN_INDEX_LSH = 6
            index_params = dict(algorithm=FLANN_INDEX_LSH, table_number=6, key_size=12, multi_probe_level=1)
            search_params = dict(checks=50)

            flann = cv2.FlannBasedMatcher(index_params, search_params)
            matches = flann.knnMatch(des1, des2, k=2)

            # Apply Lowe's ratio test
            good_matches = []
            for m, n in matches:
                if m.distance < 0.7 * n.distance:
                    good_matches.append(m)

            return good_matches

        except Exception as e:
            self.get_logger().warn(f'Feature matching failed: {str(e)}')
            return []

    def extract_matched_points(self, matches, desc1, desc2):
        """Extract matched points from feature matches"""

        kp1, _ = desc1
        kp2, _ = desc2

        if len(matches) == 0:
            return np.array([]), np.array([])

        prev_pts = np.float32([kp1[m.queryIdx].pt for m in matches]).reshape(-1, 1, 2)
        curr_pts = np.float32([kp2[m.trainIdx].pt for m in matches]).reshape(-1, 1, 2)

        return prev_pts, curr_pts

    def publish_pose_estimate(self, pose_matrix, timestamp):
        """Publish pose estimate to ROS topic"""

        pose_msg = PoseStamped()
        pose_msg.header.stamp = timestamp
        pose_msg.header.frame_id = self.map_frame

        rotation_matrix = pose_matrix[:3, :3]

        qx = (rotation_matrix[2,1] - rotation_matrix[1,2]) / (4 * qw)
        qy = (rotation_matrix[0,2] - rotation_matrix[2,0]) / (4 * qw)
        qz = (rotation_matrix[1,0] - rotation_matrix[0,1]) / (4 * qw)

        pose_msg.pose.orientation.y = float(qy)
        pose_msg.pose.orientation.z = float(qz)
        pose_msg.pose.orientation.w = float(qw)

        self.pose_publisher.publish(pose_msg)

    def publish_odometry(self, pose_matrix, timestamp):
```

```python
                    'solutions': issue_info['solutions']
                })

        potential_issues.sort(key=lambda x: x['confidence'], reverse=True)

        return {
            'diagnosis': potential_issues[:3],  # Top 3 potential issues
            'recommended_actions': potential_issues[0]['solutions'] if potential_issues else []
        }
```

```python
            "-d", "~/isaac_ros_ws/src/isaac_ros_visual_slam/rviz/visual_slam.rviz"
        ]

        process = subprocess.Popen(cmd)
        self.processes.append(process)

        time.sleep(2)  # Wait for RViz to start
        print("✅ RViz launched")

    def run_lab_experiment(self, duration=60):
        """Run the complete lab experiment"""

        print(f"🧪 Starting VSLAM lab experiment for {duration} seconds...")

        # Setup environment
        self.setup_environment()

        # Launch Isaac Sim scene
        self.launch_isaac_sim_scene()

        # Launch VSLAM pipeline
        self.launch_vslam_pipeline()

        # Launch RViz
        self.launch_rviz()

        print(f"⏳ Experiment running for {duration} seconds...")
        time.sleep(duration)

        # Collect results
        self.collect_results()

        # Cleanup
        self.cleanup()

        print("✅ VSLAM lab experiment completed!")

    def collect_results(self):
        """Collect and analyze experimental results"""

        print("📊 Collecting experimental results...")

        # Check if topics are publishing
        result = subprocess.run(
            ["ros2", "topic", "list"],
            capture_output=True,
            text=True
        )

        topics = result.stdout.split('\n')
        vslam_topics = [t for t in topics if 'visual_slam' in t or 'robot_pose' in t]

        print(f"   Found {len(vslam_topics)} VSLAM-related topics:")
        for topic in vslam_topics:
            print(f"     - {topic}")

        # Check topic statistics
        for topic in vslam_topics[:3]:  # Check first 3 topics
            if topic.strip():  # Skip empty topics
                stat_cmd = ["ros2", "topic", "hz", topic.strip()]
                stat_result = subprocess.run(stat_cmd, capture_output=True, text=True, timeout=5)
                print(f"   Topic {topic.strip()} hz: {stat_result.stdout}")

    def cleanup(self):
        """Clean up launched processes"""

        print("🧹 Cleaning up processes...")

        for process in self.processes:
            try:
                process.terminate()
                process.wait(timeout=2)
            except subprocess.TimeoutExpired:
                process.kill()

        print("✅ Cleanup complete")
```
