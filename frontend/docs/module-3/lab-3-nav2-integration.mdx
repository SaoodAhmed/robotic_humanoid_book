# Lab 3: Nav2 Integration and Bipedal Path Planning

## Overview

This lab focuses on integrating Isaac ROS VSLAM outputs with the Nav2 navigation stack for bipedal humanoid path planning. Students will learn to configure Nav2 for the unique kinematic constraints of bipedal robots and validate the complete perception-to-navigation pipeline.

## Learning Objectives

After completing this lab, students will be able to:
- Integrate Isaac ROS VSLAM outputs with Nav2 path planning
- Configure Nav2 parameters specifically for bipedal humanoid kinematics
- Validate navigation performance in simulated environments
- Troubleshoot common integration issues
- Evaluate path planning effectiveness for bipedal locomotion

## Prerequisites

Before starting this lab, ensure you have:
- Completed Lab 1: Isaac Sim Scene Creation & Synthetic Data Export
- Completed Lab 2: Isaac ROS VSLAM Pipeline
- Isaac ROS and Nav2 packages properly installed
- Isaac Sim scene with humanoid robot configured
- Working VSLAM pipeline from Lab 2
- Understanding of ROS 2 navigation concepts

## Equipment and Software Requirements

- Workstation with RTX GPU (RTX 3080 or better recommended)
- Ubuntu 22.04 LTS
- Isaac Sim 2023.1+
- Isaac ROS packages
- Nav2 packages
- CUDA 12.0+
- VRAM: Minimum 16GB (24GB+ recommended)

## Lab Duration

Estimated completion time: 3-4 hours

### Step 4: Create Bridge Node Implementation

1. Create the bridge node that handles data conversion and synchronization:
```python
#!/usr/bin/env python3
# Isaac ROS to Nav2 bridge node

import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy
from geometry_msgs.msg import PoseWithCovarianceStamped, TransformStamped
from nav_msgs.msg import OccupancyGrid, Odometry
from sensor_msgs.msg import Image, LaserScan
from tf2_ros import TransformBroadcaster
from tf2_geometry_msgs import do_transform_pose
import tf2_ros
import numpy as np
from scipy.spatial.transform import Rotation as R


class IsaacRosNav2Bridge(Node):
    def __init__(self):
        super().__init__('isaac_ros_nav2_bridge')

        # Parameters
        self.declare_parameter('max_synchronization_delay', 0.5)
        self.declare_parameter('message_queue_size', 10)
        self.declare_parameter('pose_covariance_scale', 1.0)

        # Store Isaac ROS data
        self.isaac_pose = None
        self.isaac_grid = None
        self.isaac_depth = None

        # Timers for synchronization
        self.sync_timer = self.create_timer(0.1, self.synchronize_data)

        # Subscribers for Isaac ROS data
        self.isaac_pose_sub = self.create_subscription(
            PoseWithCovarianceStamped,
            '/humanoid_robot/vslam/poses',
            self.isaac_pose_callback,
            10
        )

        self.isaac_grid_sub = self.create_subscription(
            OccupancyGrid,
            '/humanoid_robot/occupancy_grid',
            self.isaac_grid_callback,
            10
        )

        self.odom_sub = self.create_subscription(
            Odometry,
            '/odom',
            self.odom_callback,
            10
        )

        # Publishers for Nav2
        self.amcl_pose_pub = self.create_publisher(
            PoseWithCovarianceStamped,
            '/initialpose',
            10
        )

        self.global_costmap_pub = self.create_publisher(
            OccupancyGrid,
            '/global_costmap/costmap',
            10
        )

        # TF buffer and broadcaster
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)
        self.tf_broadcaster = TransformBroadcaster(self)

        # Timing for synchronization
        self.last_sync_time = self.get_clock().now()

        self.get_logger().info("Isaac ROS to Nav2 Bridge initialized")

    def isaac_pose_callback(self, msg):
        """Handle Isaac ROS pose estimation"""
        self.isaac_pose = msg
        self.get_logger().debug(f"Received Isaac ROS pose: ({msg.pose.pose.position.x:.2f}, {msg.pose.pose.position.y:.2f})")

        # Convert Isaac ROS pose to Nav2 format and publish
        nav2_pose = self.convert_pose_for_nav2(msg)
        if nav2_pose:
            self.amcl_pose_pub.publish(nav2_pose)

    def isaac_grid_callback(self, msg):
        """Handle Isaac ROS occupancy grid"""
        self.isaac_grid = msg
        self.get_logger().debug(f"Received Isaac ROS grid: {msg.info.width}x{msg.info.height}")

        # Convert Isaac ROS grid to Nav2 format and publish
        nav2_grid = self.convert_grid_for_nav2(msg)
        if nav2_grid:
            self.global_costmap_pub.publish(nav2_grid)

    def odom_callback(self, msg):
        """Handle odometry data for synchronization"""
        # Use odometry for timing synchronization
        pass

    def convert_pose_for_nav2(self, isaac_pose_msg):
        """Convert Isaac ROS pose to Nav2-compatible format"""
        try:
            # Create Nav2-compatible pose message
            nav2_pose = PoseWithCovarianceStamped()
            nav2_pose.header = isaac_pose_msg.header
            nav2_pose.header.frame_id = 'map'  # Nav2 expects map frame

            # Copy pose with potential frame transformation
            nav2_pose.pose = isaac_pose_msg.pose

            # Scale covariance if needed
            scale = self.get_parameter('pose_covariance_scale').value
            for i in range(len(nav2_pose.pose.covariance)):
                nav2_pose.pose.covariance[i] *= scale

            return nav2_pose
        except Exception as e:
            self.get_logger().error(f"Error converting pose: {str(e)}")
            return None

    def convert_grid_for_nav2(self, isaac_grid_msg):
        """Convert Isaac ROS grid to Nav2-compatible format"""
        try:
            # Create Nav2-compatible occupancy grid
            nav2_grid = OccupancyGrid()
            nav2_grid.header = isaac_grid_msg.header
            nav2_grid.header.frame_id = 'map'  # Nav2 expects map frame

            nav2_grid.info = isaac_grid_msg.info
            nav2_grid.data = isaac_grid_msg.data

            return nav2_grid
        except Exception as e:
            self.get_logger().error(f"Error converting grid: {str(e)}")
            return None

    def synchronize_data(self):
        """Synchronize Isaac ROS and Nav2 data streams"""
        current_time = self.get_clock().now()
        max_delay = self.get_parameter('max_synchronization_delay').value

        # Check if Isaac ROS data is recent enough
        if self.isaac_pose is not None:
            pose_age = (current_time - rclpy.time.Time.from_msg(self.isaac_pose.header.stamp)).nanoseconds / 1e9
            if pose_age > max_delay:
                self.isaac_pose = None  # Invalidate old data
                self.get_logger().warn("Isaac ROS pose data is too old, invalidating")

        if self.isaac_grid is not None:
            grid_age = (current_time - rclpy.time.Time.from_msg(self.isaac_grid.header.stamp)).nanoseconds / 1e9
            if grid_age > max_delay:
                self.isaac_grid = None  # Invalidate old data
                self.get_logger().warn("Isaac ROS grid data is too old, invalidating")

        # Log synchronization status
        if self.isaac_pose is not None and self.isaac_grid is not None:
            self.get_logger().info("Isaac ROS and Nav2 data synchronized successfully")
        elif self.isaac_pose is not None:
            self.get_logger().info("Isaac ROS pose available, waiting for grid")
        elif self.isaac_grid is not None:
            self.get_logger().info("Isaac ROS grid available, waiting for pose")
        else:
            self.get_logger().info("Waiting for Isaac ROS data...")

    def validate_integration(self):
        """Validate that Isaac ROS and Nav2 are properly integrated"""
        # Check if both systems are publishing data
        isaac_active = self.isaac_pose is not None or self.isaac_grid is not None
        # In a real implementation, we would check Nav2 status as well

        if isaac_active:
            self.get_logger().info("Isaac ROS to Nav2 integration validated successfully")
            return True
        else:
            self.get_logger().warn("Isaac ROS to Nav2 integration validation failed - no data received")
            return False


def main(args=None):
    rclpy.init(args=args)

    bridge = IsaacRosNav2Bridge()

    # Validate integration periodically
    validation_timer = bridge.create_timer(5.0, bridge.validate_integration)

    try:
        rclpy.spin(bridge)
    except KeyboardInterrupt:
        bridge.get_logger().info("Bridge interrupted by user")
    finally:
        bridge.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

            start_pt = path[i]
            end_pt = path[i + 1]

            # Calculate direction vector
            dx = end_pt[0] - start_pt[0]
            dy = end_pt[1] - start_pt[1]

            for j in range(num_steps):

                step_y = start_pt[1] + t * dy

                if (i + j) % 2 == 0:
                    foot_offset = self.step_width / 2
                else:
                    foot_offset = -self.step_width / 2

                # Apply lateral offset for alternating feet
                angle = math.atan2(dy, dx)
                step_x += foot_offset * math.sin(angle)
                step_y -= foot_offset * math.cos(angle)

                footsteps.append((step_x, step_y, angle))

        self.isaac_pose_times.append(current_time)

        # Calculate frequency if we have enough data
        if len(self.isaac_pose_times) > 1:
            time_diffs = [self.isaac_pose_times[i] - self.isaac_pose_times[i-1]
                         for i in range(1, len(self.isaac_pose_times))]
            if time_diffs:
                avg_period = sum(time_diffs) / len(time_diffs)
                freq = 1.0 / avg_period if avg_period > 0 else 0.0

                freq_msg = Float32()
                freq_msg.data = freq
                self.pose_freq_pub.publish(freq_msg)

                self.performance_metrics['isaac_pose_frequency'].append(freq)

    def nav_path_callback(self, msg):
        """Track Nav2 path planning timing"""
        current_time = time.time()
        self.nav_path_times.append(current_time)

        if len(self.nav_path_times) > 1:
            time_diffs = [self.nav_path_times[i] - self.nav_path_times[i-1]
                         for i in range(1, len(self.nav_path_times))]
            if time_diffs:
                avg_period = sum(time_diffs) / len(time_diffs)
                freq = 1.0 / avg_period if avg_period > 0 else 0.0

                freq_msg = Float32()
                freq_msg.data = freq
                self.path_freq_pub.publish(freq_msg)

                self.performance_metrics['nav_path_frequency'].append(freq)

    def costmap_callback(self, msg):
        """Track costmap update timing"""
        current_time = time.time()
        self.costmap_times.append(current_time)

    def scan_callback(self, msg):
        """Track sensor data timing"""
        current_time = time.time()
        self.scan_times.append(current_time)

    def perform_analysis(self):
        """Perform periodic performance analysis"""
        # Calculate current frequencies
        if len(self.isaac_pose_times) > 1:
            isaac_freq = self.calculate_frequency(self.isaac_pose_times)
            self.get_logger().debug(f"Isaac ROS Pose Frequency: {isaac_freq:.2f} Hz")

        if len(self.nav_path_times) > 1:
            nav_freq = self.calculate_frequency(self.nav_path_times)
            self.get_logger().debug(f"Nav2 Path Frequency: {nav_freq:.2f} Hz")

        if len(self.costmap_times) > 1:
            costmap_freq = self.calculate_frequency(self.costmap_times)
            self.get_logger().debug(f"Costmap Update Frequency: {costmap_freq:.2f} Hz")

        # Monitor system resources
        cpu_usage = self.get_cpu_usage()
        mem_usage = self.get_memory_usage()

        cpu_msg = Float32()
        cpu_msg.data = cpu_usage
        self.cpu_usage_pub.publish(cpu_msg)

        mem_msg = Float32()
        mem_msg.data = mem_usage
        self.memory_usage_pub.publish(mem_msg)

        self.performance_metrics['cpu_usage'].append(cpu_usage)
        self.performance_metrics['memory_usage'].append(mem_usage)

    def calculate_frequency(self, time_list):
        """Calculate frequency from a list of timestamps"""
        if len(time_list) < 2:
            return 0.0

        time_diffs = [time_list[i] - time_list[i-1] for i in range(1, len(time_list))]
        if not time_diffs:
            return 0.0

        avg_period = sum(time_diffs) / len(time_diffs)
        return 1.0 / avg_period if avg_period > 0 else 0.0

    def get_cpu_usage(self):
        """Get current CPU usage"""
        import psutil
        return psutil.cpu_percent(interval=None)

    def get_memory_usage(self):
        """Get current memory usage"""
        import psutil
        memory = psutil.virtual_memory()
        return memory.percent

    def report_performance(self):
        """Report current performance metrics"""
        self.get_logger().info("--- Performance Analysis Report ---")

        if self.performance_metrics['isaac_pose_frequency']:
            avg_freq = statistics.mean(self.performance_metrics['isaac_pose_frequency'])
            self.get_logger().info(f"Average Isaac ROS Pose Frequency: {avg_freq:.2f} Hz")

        if self.performance_metrics['nav_path_frequency']:
            avg_freq = statistics.mean(self.performance_metrics['nav_path_frequency'])
            self.get_logger().info(f"Average Nav2 Path Frequency: {avg_freq:.2f} Hz")

        if self.performance_metrics['cpu_usage']:
            avg_cpu = statistics.mean(self.performance_metrics['cpu_usage'])
            max_cpu = max(self.performance_metrics['cpu_usage'])
            self.get_logger().info(f"Average CPU Usage: {avg_cpu:.2f}% (Max: {max_cpu:.2f}%)")

        if self.performance_metrics['memory_usage']:
            avg_mem = statistics.mean(self.performance_metrics['memory_usage'])
            max_mem = max(self.performance_metrics['memory_usage'])
            self.get_logger().info(f"Average Memory Usage: {avg_mem:.2f}% (Max: {max_mem:.2f}%)")

        # Check for performance issues
        self.check_performance_issues()

        self.get_logger().info("-------------------------------")

    def check_performance_issues(self):
        """Check for performance bottlenecks"""
        # Check if Isaac ROS pose frequency is too low
        if (self.performance_metrics['isaac_pose_frequency'] and
            statistics.mean(self.performance_metrics['isaac_pose_frequency']) < 5.0):
            self.get_logger().warn("Isaac ROS pose frequency is low (< 5 Hz)")

        # Check if Nav2 path frequency is too low
        if (self.performance_metrics['nav_path_frequency'] and
            statistics.mean(self.performance_metrics['nav_path_frequency']) < 1.0):
            self.get_logger().warn("Nav2 path frequency is low (< 1 Hz)")

        # Check if CPU usage is too high
        if (self.performance_metrics['cpu_usage'] and
            statistics.mean(self.performance_metrics['cpu_usage']) > 80.0):
            self.get_logger().warn("CPU usage is high (> 80%)")

        # Check if memory usage is too high
        if (self.performance_metrics['memory_usage'] and
            statistics.mean(self.performance_metrics['memory_usage']) > 85.0):
            self.get_logger().warn("Memory usage is high (> 85%)")

    def generate_performance_report(self, filename=None):
        """Generate a detailed performance report"""
        if filename is None:
            from datetime import datetime
            filename = f"/tmp/performance_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"

        with open(filename, 'w') as f:
            f.write("Isaac ROS-Nav2 Integration Performance Report\n")
            f.write("=" * 50 + "\n\n")

            f.write("Performance Metrics:\n")
            for metric, values in self.performance_metrics.items():
                if values:
                    f.write(f"  {metric}:\n")
                    f.write(f"    Count: {len(values)}\n")
                    f.write(f"    Average: {statistics.mean(values):.3f}\n")
                    f.write(f"    Min: {min(values):.3f}\n")
                    f.write(f"    Max: {max(values):.3f}\n")
                    if len(values) > 1:
                        f.write(f"    Std Dev: {statistics.stdev(values):.3f}\n")
                    f.write("\n")

        self.get_logger().info(f"Detailed performance report saved to {filename}")

    def plot_performance_metrics(self):
        """Plot performance metrics over time"""
        try:

            fig.suptitle('Isaac ROS-Nav2 Performance Analysis')

                axes[1, 1].set_title('Memory Usage')

            plt.tight_layout()

            plot_filename = f"/tmp/performance_plot_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png"
            plt.savefig(plot_filename)
            self.get_logger().info(f"Performance plot saved to {plot_filename}")

        except ImportError:
            self.get_logger().warn("matplotlib not available, skipping plot generation")


def main(args=None):

    analyzer = PerformanceAnalyzer()

    # Run for a specific duration (e.g., 5 minutes) then generate report
    import threading
    import time

    def run_for_duration():
