    def check_performance_thresholds(self):
        """Check if performance metrics meet thresholds"""

        current_metrics = {key: values[-1] if values else 0
                          for key, values in self.metrics.items()}

        violations = []
        for metric, threshold in self.thresholds.items():
            if metric in current_metrics:
                if metric.startswith('max_') and current_metrics[metric.replace('max_', '')] > threshold:
                    violations.append(f"{metric.replace('max_', '')} exceeds threshold")
                elif metric == 'min_fps' and current_metrics['fps'] < threshold:
                    violations.append("FPS below minimum threshold")

        return violations, current_metrics

    def generate_performance_report(self):
        """Generate comprehensive performance report"""

        if not self.metrics['fps']:
            return "No performance data collected"

        avg_fps = sum(self.metrics['fps']) / len(self.metrics['fps']) if self.metrics['fps'] else 0
        min_fps = min(self.metrics['fps']) if self.metrics['fps'] else 0
        max_fps = max(self.metrics['fps']) if self.metrics['fps'] else 0

        report = {
            'average_fps': avg_fps,
            'minimum_fps': min_fps,
            'maximum_fps': max_fps,
            'target_met': avg_fps >= 10,
            'total_frames': len(self.metrics['fps']),
            'violations': self.check_performance_thresholds()[0],
            'memory_trend': 'increasing' if len(self.metrics['memory_usage']) > 1 and
                           self.metrics['memory_usage'][-1] > self.metrics['memory_usage'][0] else 'stable',
            'gpu_trend': 'increasing' if len(self.metrics['gpu_usage']) > 1 and
                         self.metrics['gpu_usage'][-1] > self.metrics['gpu_usage'][0] else 'stable'
        }

        return report

## Troubleshooting Complex Scene Performance

### Common Complex Scene Issues

#### Multi-Sensor Bottlenecks
- **Symptoms**: Performance degradation with multiple sensors
- **Solutions**:
  - Implement sensor scheduling
  - Reduce sensor update frequency
  - Use shared rendering pipelines
  - Optimize sensor-specific calculations

#### Physics Complexity Issues
- **Symptoms**: Physics simulation impacting frame rate
- **Solutions**:
  - Implement physics LOD
  - Reduce solver iterations for distant objects
  - Use simplified collision shapes
  - Apply temporal sub-stepping

#### Memory Pressure
- **Symptoms**: Slow performance, potential crashes due to memory
- **Solutions**:
  - Enable texture streaming
  - Implement geometry streaming
  - Use resource pooling
  - Apply aggressive caching
