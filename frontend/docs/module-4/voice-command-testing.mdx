# Voice Command Testing Utilities

## Overview
This document describes the testing utilities for the voice command processing system. These utilities enable comprehensive testing of the speech recognition pipeline, including automated testing, performance measurement, and validation of recognition accuracy.

## Core Testing Components

### 1. Audio Simulation
The testing framework includes utilities to simulate various audio conditions for consistent testing:

```python
import numpy as np
import wave
import struct
import os
from typing import List, Tuple, Optional

class AudioSimulator:
    def __init__(self, sample_rate=16000):
        self.sample_rate = sample_rate

    def generate_sine_wave(self, frequency: float, duration: float, amplitude: float = 0.5) -> np.ndarray:
        """
        Generate a sine wave for testing purposes
        """
        t = np.linspace(0, duration, int(self.sample_rate * duration), endpoint=False)
        wave_data = amplitude * np.sin(2 * np.pi * frequency * t)
        return wave_data

    def generate_speech_like_signal(self, text: str, duration: float) -> np.ndarray:
        """
        Generate a speech-like signal for testing (simplified)
        """
        # In a real implementation, this would use more sophisticated speech synthesis
        # For testing, we'll create a signal with multiple frequencies that change over time
        t = np.linspace(0, duration, int(self.sample_rate * duration), endpoint=False)

        # Create a signal with multiple harmonics that change over time
        signal = np.zeros_like(t)
        for i, char in enumerate(text[:10]):  # Use first 10 characters to influence frequencies
            freq = 200 + ord(char) % 100  # Map character to frequency range 200-300 Hz
            start_idx = int(i * len(t) / 10)
            end_idx = int((i + 1) * len(t) / 10)
            if end_idx > len(t):
                end_idx = len(t)

            segment_t = t[start_idx:end_idx]
            signal[start_idx:end_idx] += 0.3 * np.sin(2 * np.pi * freq * segment_t)

        return signal

    def add_noise(self, signal: np.ndarray, snr_db: float) -> np.ndarray:
        """
        Add noise to a signal at a specific signal-to-noise ratio
        """
        # Calculate signal power
        signal_power = np.mean(signal ** 2)

        # Calculate noise power from SNR
        snr_linear = 10 ** (snr_db / 10)
        noise_power = signal_power / snr_linear

        # Generate noise
        noise = np.random.normal(0, np.sqrt(noise_power), signal.shape)

        # Add noise to signal
        noisy_signal = signal + noise

        return noisy_signal

    def save_audio_file(self, signal: np.ndarray, filename: str):
        """
        Save a signal to a WAV file
        """
        # Ensure signal is in the range [-1, 1]
        signal = np.clip(signal, -1, 1)

        # Convert to 16-bit integers
        signal_int16 = (signal * 32767).astype(np.int16)

        # Write to WAV file
        with wave.open(filename, 'w') as wav_file:
            wav_file.setnchannels(1)  # Mono
            wav_file.setsampwidth(2)  # 16-bit
            wav_file.setframerate(self.sample_rate)
            wav_file.writeframes(signal_int16.tobytes())

    def load_audio_file(self, filename: str) -> Tuple[np.ndarray, int]:
        """
        Load an audio file and return the signal and sample rate
        """
        with wave.open(filename, 'r') as wav_file:
            frames = wav_file.readframes(wav_file.getnframes())
            sample_rate = wav_file.getframerate()
            signal = np.frombuffer(frames, dtype=np.int16).astype(np.float32) / 32767.0
            return signal, sample_rate
```

### 2. Recognition Testing Framework

```python
import unittest
from typing import Dict, Any
import time
import logging

class VoiceCommandTester:
    def __init__(self, voice_processor):
        self.voice_processor = voice_processor
        self.test_results = []

    def test_recognition_accuracy(self, test_cases: List[Dict[str, Any]]) -> Dict[str, float]:
        """
        Test recognition accuracy with predefined audio files
        """
        correct = 0
        total = len(test_cases)

        for test_case in test_cases:
            expected_text = test_case['expected_text']
            audio_file = test_case['audio_file']
            confidence_threshold = test_case.get('confidence_threshold', 0.5)

            # Simulate recognition process
            recognized_text, confidence = self._simulate_recognition(audio_file)

            # Check if recognition matches expected text
            if self._compare_texts(recognized_text, expected_text) and confidence >= confidence_threshold:
                correct += 1
                test_result = {
                    'status': 'PASS',
                    'expected': expected_text,
                    'actual': recognized_text,
                    'confidence': confidence,
                    'audio_file': audio_file
                }
            else:
                test_result = {
                    'status': 'FAIL',
                    'expected': expected_text,
                    'actual': recognized_text,
                    'confidence': confidence,
                    'audio_file': audio_file
                }

            self.test_results.append(test_result)

        accuracy = correct / total if total > 0 else 0
        return {
            'accuracy': accuracy,
            'correct': correct,
            'total': total,
            'results': self.test_results
        }

    def test_latency_performance(self, iterations: int = 10) -> Dict[str, float]:
        """
        Test the latency of the recognition system
        """
        latencies = []

        for i in range(iterations):
            # Generate a test signal
            test_signal = self.audio_sim.generate_speech_like_signal("test command", 2.0)

            # Measure processing time
            start_time = time.time()
            # Simulate processing the signal
            result = self._simulate_recognition_from_signal(test_signal)
            end_time = time.time()

            latency = (end_time - start_time) * 1000  # Convert to milliseconds
            latencies.append(latency)

        return {
            'average_latency': sum(latencies) / len(latencies),
            'min_latency': min(latencies),
            'max_latency': max(latencies),
            'p95_latency': sorted(latencies)[int(0.95 * len(latencies))],
            'latencies': latencies
        }

    def test_noise_tolerance(self, base_snr_db: float = 20, step_db: float = 5) -> Dict[str, Any]:
        """
        Test recognition performance at various noise levels
        """
        results = {}

        for snr_db in range(base_snr_db, -1, -step_db):
            # Generate a clean signal
            clean_signal = self.audio_sim.generate_speech_like_signal("go to kitchen", 2.0)

            # Add noise
            noisy_signal = self.audio_sim.add_noise(clean_signal, snr_db)

            # Test recognition
            recognized_text, confidence = self._simulate_recognition_from_signal(noisy_signal)

            results[snr_db] = {
                'recognized_text': recognized_text,
                'confidence': confidence,
                'expected': 'go to kitchen'
            }

        return results

    def _simulate_recognition(self, audio_file: str) -> Tuple[str, float]:
        """
        Simulate the recognition process (in a real implementation, this would call the actual STT system)
        """
        # In a real implementation, this would load the audio file and process it with the STT engine
        # For this simulation, we'll return a result based on the filename
        if "kitchen" in audio_file.lower():
            return "go to kitchen", 0.85
        elif "ball" in audio_file.lower():
            return "find the red ball", 0.78
        else:
            return "unknown command", 0.3

    def _simulate_recognition_from_signal(self, signal: np.ndarray) -> Tuple[str, float]:
        """
        Simulate recognition from a signal array
        """
        # In a real implementation, this would process the signal with the STT engine
        # For this simulation, we'll return a fixed result
        return "test command", 0.82

    def _compare_texts(self, text1: str, text2: str, tolerance: float = 0.8) -> bool:
        """
        Compare two texts for similarity
        """
        # Simple word-based comparison
        words1 = set(text1.lower().split())
        words2 = set(text2.lower().split())

        if not words1 or not words2:
            return text1.lower() == text2.lower()

        # Calculate Jaccard similarity
        intersection = len(words1.intersection(words2))
        union = len(words1.union(words2))

        similarity = intersection / union if union > 0 else 0
        return similarity >= tolerance

    def generate_test_report(self) -> str:
        """
        Generate a comprehensive test report
        """
        report = []
        report.append("Voice Command Processing Test Report")
        report.append("=" * 40)

        # Accuracy results
        passed = sum(1 for r in self.test_results if r['status'] == 'PASS')
        total = len(self.test_results)
        accuracy = passed / total if total > 0 else 0

        report.append(f"Recognition Accuracy: {accuracy:.2%} ({passed}/{total})")
        report.append("")

        # Detailed results
        report.append("Detailed Results:")
        for i, result in enumerate(self.test_results):
            status = "✓" if result['status'] == 'PASS' else '✗'
            report.append(f"{i+1}. {status} Expected: '{result['expected']}', Actual: '{result['actual']}', Confidence: {result['confidence']:.2f}")

        return "\n".join(report)
```

```python
        self.test_audio_file = os.path.join(self.test_dir, "test_kitchen.wav")

        """Clean up after each test method."""
        # Remove temporary files

        shutil.rmtree(self.test_dir)

        """Test that audio generation works correctly."""
        signal = self.audio_sim.generate_sine_wave(440, 1.0)  # 440Hz for 1 second
        self.assertEqual(len(signal), 16000)  # 16000 samples at 16kHz

        # Check that signal values are in reasonable range
        self.assertTrue(-1 <= signal.min() <= 1)
        self.assertTrue(-1 <= signal.max() <= 1)

    def test_noise_addition(self):
        """Test that noise addition works correctly."""
        clean_signal = self.audio_sim.generate_sine_wave(440, 1.0)
        noisy_signal = self.audio_sim.add_noise(clean_signal, 20)  # 20dB SNR

        # The noisy signal should be different from the clean signal
        self.assertFalse(np.array_equal(clean_signal, noisy_signal))
```

## Performance Testing Utilities

### 1. Load Testing

```python
import threading
import time
from concurrent.futures import ThreadPoolExecutor
import statistics

class LoadTester:
    def __init__(self, voice_processor, num_threads=5):
        self.voice_processor = voice_processor
        self.num_threads = num_threads
        self.results = []

    def simulate_concurrent_users(self, num_users: int, test_duration: int = 60) -> Dict[str, Any]:
        """
        Simulate multiple concurrent users using the voice recognition system
        """
        start_time = time.time()
        end_time = start_time + test_duration

        def user_simulation(user_id: int):
            user_results = []
            audio_sim = AudioSimulator()

            while time.time() < end_time:
                # Generate a random command
                commands = ["go to kitchen", "find the red ball", "pick up the cup", "move to the living room"]
                import random
                command = random.choice(commands)

                # Generate audio for the command
                signal = audio_sim.generate_speech_like_signal(command, 1.5)

                # Measure processing time
                process_start = time.time()
                # In a real implementation, this would process the audio
                recognized_text, confidence = self._simulate_recognition_from_signal(signal)
                process_end = time.time()

                user_results.append({
                    'user_id': user_id,
                    'command': command,
                    'recognized': recognized_text,
                    'confidence': confidence,
                    'processing_time': process_end - process_start,
                    'timestamp': time.time()
                })

                # Wait a bit before next command
                time.sleep(2)

            return user_results

        # Execute user simulations in parallel
        with ThreadPoolExecutor(max_workers=self.num_threads) as executor:
            futures = [executor.submit(user_simulation, i) for i in range(num_users)]
            all_results = [future.result() for future in futures]

        # Flatten results
        flattened_results = [item for sublist in all_results for item in sublist]

        # Calculate statistics
        processing_times = [r['processing_time'] for r in flattened_results]

        return {
            'total_requests': len(flattened_results),
            'average_processing_time': statistics.mean(processing_times) if processing_times else 0,
            'p95_processing_time': sorted(processing_times)[int(0.95 * len(processing_times))] if processing_times else 0,
            'max_processing_time': max(processing_times) if processing_times else 0,
            'results': flattened_results
        }
```

### 2. Stress Testing

```python
class StressTester:
    def __init__(self, voice_processor):
        self.voice_processor = voice_processor

    def stress_test(self, max_load_factor: float = 2.0, test_duration: int = 300) -> Dict[str, Any]:
        """
        Perform stress testing by gradually increasing load until system fails
        """
        start_time = time.time()
        results = []

        # Start with a low load and gradually increase
        current_load = 1.0
        step_size = 0.1

        while current_load <= max_load_factor and (time.time() - start_time) < test_duration:
            # Calculate number of concurrent users based on load factor
            num_users = int(current_load * 10)  # Base of 10 users at 1.0 load factor

            print(f"Testing at {current_load:.1f}x load ({num_users} users)")

            # Run load test at current load level
            load_tester = LoadTester(self.voice_processor)
            test_results = load_tester.simulate_concurrent_users(num_users, 30)  # 30 second test

            # Check if system is still performing adequately
            avg_processing_time = test_results['average_processing_time']
            p95_processing_time = test_results['p95_processing_time']

            results.append({
                'load_factor': current_load,
                'num_users': num_users,
                'avg_processing_time': avg_processing_time,
                'p95_processing_time': p95_processing_time,
                'total_requests': test_results['total_requests'],
                'success': avg_processing_time < 1.0 and p95_processing_time < 2.0  # Thresholds in seconds
            })

            # If system is still performing well, increase load
            if test_results['average_processing_time'] < 1.0:  # Less than 1 second average
                current_load += step_size
            else:
                # System is struggling, reduce step size and approach more carefully
                step_size = 0.05
                current_load += step_size

            time.sleep(5)  # Brief pause between load levels

        return results
```
