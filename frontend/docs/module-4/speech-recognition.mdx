# Speech Recognition Utilities

## Overview
This document describes the utilities for speech recognition in the Vision-Language-Action (VLA) system. The speech recognition component converts spoken commands into structured text that can be processed by the LLM and action execution system.

## Core Components

### Speech Recognition Engine
The system uses the SpeechRecognition library with multiple backends:
- **Google Web Speech API**: High accuracy for online recognition
- **PocketSphinx**: Offline recognition capability
- **Vosk**: High-quality offline recognition
- **Wit.ai**: Alternative cloud-based recognition

### Configuration Parameters
- **Sample Rate**: 16000 Hz (configurable)
- **Chunk Size**: 1024 (configurable)
- **Energy Threshold**: 4000 (auto-adjusting)
- **Pause Threshold**: 0.8 seconds
- **Phrase Threshold**: 0.3 seconds

import speech_recognition as sr
import pyaudio
import threading
import time
import logging
from typing import Optional, Callable

class SpeechRecognitionManager:
    def __init__(self, config):
        self.config = config
        self.recognizer = sr.Recognizer()
        self.microphone = sr.Microphone()
        self.is_listening = False
        self.is_active = False

        # Configure recognition parameters
        self.setup_recognition_parameters()

        # Initialize audio engine
        self.audio_engine = pyaudio.PyAudio()

    def setup_recognition_parameters(self):
        """
        Configure the speech recognition parameters
        """
        # Set energy threshold for silence detection
        self.recognizer.energy_threshold = self.config.get('energy_threshold', 4000)

        # Enable dynamic energy threshold adjustment
        self.recognizer.dynamic_energy_threshold = True

        # Set pause threshold (seconds of silence after a phrase)
        self.recognizer.pause_threshold = self.config.get('pause_threshold', 0.8)

        # Set phrase threshold (minimum seconds for a phrase)
        self.recognizer.phrase_threshold = self.config.get('phrase_threshold', 0.3)

        # Set minimum seconds for an empty phrase
        self.recognizer.non_speaking_duration = self.config.get('non_speaking_duration', 0.5)

    def calibrate_microphone(self):
        """
        Calibrate the microphone for ambient noise
        """
        with self.microphone as source:
            logging.info("Calibrating microphone for ambient noise...")
            self.recognizer.adjust_for_ambient_noise(source, duration=1.0)
            logging.info("Microphone calibration complete")

    def start_listening(self, callback: Callable[[str, float], None],
                       activation_phrase: str = "hey robot"):
        """
        Start listening for voice commands
        """
        self.is_active = True
        self.calibrate_microphone()

        logging.info(f"Starting voice command listener with activation phrase: '{activation_phrase}'")

        while self.is_active:
            try:
                with self.microphone as source:
                    # Listen for audio
                    audio = self.recognizer.listen(source, timeout=1.0, phrase_time_limit=5.0)

                    # Recognize speech
                    text = self.recognize_speech(audio)

                    if text:
                        # Calculate confidence score
                        confidence = self.estimate_confidence(text)

                        # Check if this is an activation command
                        if self.is_activation_command(text, activation_phrase):
                            # Listen for the actual command
                            command_text = self.listen_for_command()
                            if command_text:
                                command_confidence = self.estimate_confidence(command_text)
                                callback(command_text, command_confidence)
                        else:
                            # Check if the entire phrase is a command (no activation needed)
                            callback(text, confidence)

            except sr.WaitTimeoutError:
                # No speech detected, continue listening
                continue
            except sr.UnknownValueError:
                # Speech not understood, continue listening
                continue
            except sr.RequestError as e:
                logging.error(f"Speech recognition error: {e}")
                # Continue listening but log the error
                continue
            except KeyboardInterrupt:
                logging.info("Speech recognition interrupted by user")
                break

    def recognize_speech(self, audio) -> Optional[str]:
        """
        Recognize speech from audio using the configured engine
        """
        try:
            # Determine which recognition engine to use
            engine = self.config.get('stt_engine', 'google')

            if engine == 'google':
                text = self.recognizer.recognize_google(audio)
            elif engine == 'google_cloud':
                text = self.recognizer.recognize_google_cloud(audio,
                    credentials_json=self.config.get('google_cloud_credentials'))
            elif engine == 'sphinx':
                text = self.recognizer.recognize_sphinx(audio)
            elif engine == 'vosk':
                # Vosk requires special handling
                text = self.recognize_vosk(audio)
            elif engine == 'wit':
                text = self.recognizer.recognize_wit(audio,
                    key=self.config.get('wit_ai_key'))
            else:
                # Default to Google
                text = self.recognizer.recognize_google(audio)

            return text

        except sr.UnknownValueError:
            logging.warning("Speech recognition could not understand audio")
            return None
        except sr.RequestError as e:
            logging.error(f"Speech recognition request error: {e}")
            return None

    def recognize_vosk(self, audio) -> Optional[str]:
        """
        Recognize speech using Vosk engine
        """
        # This is a simplified implementation
        # In a real implementation, you would use the Vosk library directly
        try:
            # Convert audio to appropriate format for Vosk
            audio_data = audio.get_raw_data(convert_rate=16000, convert_width=2)

            # In a real implementation, you would use:
            # import vosk
            # model = vosk.Model("path/to/model")
            # rec = vosk.Recognizer(model, 16000)
            # if rec.AcceptWaveform(audio_data):
            #     result = rec.Result()
            #     return json.loads(result)['text']

            # For this implementation, we'll fall back to Google
            return self.recognizer.recognize_google(audio)
        except Exception as e:
            logging.error(f"Vosk recognition error: {e}")
            return None

    def is_activation_command(self, text: str, activation_phrase: str) -> bool:
        """
        Check if the text contains the activation phrase
        """
        text_lower = text.lower()
        activation_lower = activation_phrase.lower()

        # Check for exact match or close match (with some tolerance for speech errors)
        if activation_lower in text_lower:
            return True

        # Additional fuzzy matching could be implemented here
        # For now, we'll use a simple similarity check
        return self.calculate_similarity(text_lower, activation_lower) > 0.7

    def calculate_similarity(self, str1: str, str2: str) -> float:
        """
        Calculate similarity between two strings (simplified)
        """
        # This is a basic implementation; in a real system you might use
        # difflib.SequenceMatcher or a more sophisticated algorithm
        try:
            from difflib import SequenceMatcher
            return SequenceMatcher(None, str1, str2).ratio()
        except ImportError:
            # Fallback if difflib is not available
            min_len = min(len(str1), len(str2))
            if min_len == 0:
                return 0.0

            matches = sum(1 for a, b in zip(str1, str2) if a == b)
            return matches / min_len

    def listen_for_command(self) -> Optional[str]:
        """
        Listen for the actual command after activation
        """
        try:
            with self.microphone as source:
                logging.info("Listening for command after activation...")
                # Listen for the next phrase (the actual command)
                audio = self.recognizer.listen(source, timeout=5.0, phrase_time_limit=10.0)

                command_text = self.recognize_speech(audio)
                return command_text
        except sr.WaitTimeoutError:
            logging.warning("Timeout waiting for command after activation")
            return None
        except Exception as e:
            logging.error(f"Error listening for command: {e}")
            return None

    def estimate_confidence(self, text: str) -> float:
        """
        Estimate confidence for recognized text (simplified)
        """
        # In a real implementation, this would use more sophisticated methods
        # based on the recognition engine's confidence scores

        # For now, we'll return a confidence based on text length and structure
        if not text or len(text.strip()) < 3:
            return 0.0

        # Longer, more structured text gets higher confidence
        confidence = min(0.5 + len(text) * 0.02, 1.0)

        # Adjust based on common speech recognition patterns
        if '?' in text or '!' in text:
            confidence += 0.1
        elif text.lower().startswith(('go', 'move', 'navigate', 'find', 'grasp', 'pick', 'bring')):
            confidence += 0.1

        return min(confidence, 1.0)

    def stop_listening(self):
        """
        Stop the speech recognition process
        """
        self.is_active = False
        logging.info("Speech recognition stopped")

    def test_microphone(self) -> bool:
        """
        Test if microphone is working properly
        """
        try:
            with self.microphone as source:
                # Adjust for ambient noise
                self.recognizer.adjust_for_ambient_noise(source, duration=0.5)

                # Try to listen for a short period
                audio = self.recognizer.listen(source, timeout=1.0)

                # If we get here, microphone is working
                return True
        except Exception as e:
            logging.error(f"Microphone test failed: {e}")
            return False
