# Docusaurus Build Validation for MDX Formatting and Navigation

## Overview

This document outlines the procedures for validating that the Vision-Language-Action (VLA) module content builds correctly with Docusaurus, ensuring proper MDX formatting, navigation structure, and overall site integrity. The validation process confirms that all content renders correctly and maintains the expected educational experience.

### Common MDX Issues to Validate

1. **Frontmatter Validation**
   - Ensure all MDX files have proper frontmatter
   - Verify `sidebar_position` is set appropriately
   - Check that titles are properly formatted

2. **Import Statement Validation**
   - Verify all Docusaurus-specific imports are correct
   - Check that component imports are properly formatted
   - Validate that imports are only at the top of the file

3. **Code Block Validation**
   - Ensure triple backticks are properly closed
   - Verify language identifiers are correct
   - Check for nested code blocks that might break parsing

4. **Link Validation**
   - Confirm relative links point to existing files
   - Verify anchor links are properly formatted
   - Check that external links are valid

## Automated Validation Scripts

### Build Validation Script

```javascript
// scripts/validate-build.js
const fs = require('fs');
const path = require('path');
const { execSync } = require('child_process');

class BuildValidator {
  constructor(docsPath = './docs/module-4/') {
    this.docsPath = docsPath;
    this.results = {
      mdxSyntax: [],
      navigation: [],
      links: [],
      buildSuccess: false
    };
  }

  async validateAll() {
    console.log('Starting Docusaurus build validation...');

    // Validate MDX syntax
    await this.validateMdxSyntax();

    // Validate navigation structure
    await this.validateNavigation();

    // Validate internal links
    await this.validateInternalLinks();

    // Test build process
    await this.testBuild();

    return this.results;
  }

  async validateMdxSyntax() {
    console.log('Validating MDX syntax...');

    const mdxFiles = this.getMdxFiles();

    for (const file of mdxFiles) {
      const content = fs.readFileSync(file, 'utf8');

      // Check for proper frontmatter
      const hasFrontmatter = this.hasProperFrontmatter(content);

      // Check for proper code block closure
      const hasProperCodeBlocks = this.hasProperCodeBlocks(content);

      // Check for import statements at top
      const hasProperImports = this.hasProperImports(content);

      this.results.mdxSyntax.push({
        file: file,
        valid: hasFrontmatter && hasProperCodeBlocks && hasProperImports,
        issues: [
          ...(hasFrontmatter ? [] : ['Missing or malformed frontmatter']),
          ...(hasProperCodeBlocks ? [] : ['Improperly closed code blocks']),
          ...(hasProperImports ? [] : ['Improper import placement'])
        ]
      });
    }

    console.log(`Validated ${mdxFiles.length} MDX files`);
  }

  getMdxFiles() {
    const files = [];
    const walk = (dir) => {
      const dirItems = fs.readdirSync(dir);
      for (const item of dirItems) {
        const fullPath = path.join(dir, item);
        const stat = fs.statSync(fullPath);

        if (stat.isDirectory()) {
          walk(fullPath);
        } else if (item.endsWith('.mdx')) {
          files.push(fullPath);
        }
      }
    };

    walk(this.docsPath);
    return files;
  }

  hasProperFrontmatter(content) {
    const lines = content.split('\n');
    return lines[0] === '---' && lines.slice(1).some(line => line.trim() === '---');
  }

  hasProperCodeBlocks(content) {
    // Count opening and closing triple backticks
    const openTicks = (content.match(/```/g) || []).length;
    return openTicks % 2 === 0; // Should be even number (open/close pairs)
  }

  hasProperImports(content) {
    // Imports should be at the top, before any content
    const lines = content.split('\n');
    let contentStarted = false;

    for (const line of lines) {
      const trimmed = line.trim();

      if (!trimmed.startsWith('import ') && !trimmed.startsWith('---') && trimmed !== '') {
        contentStarted = true;
      }

      if (contentStarted && trimmed.startsWith('import ')) {
        return false; // Import found after content started
      }
    }

    return true;
  }

  async validateNavigation() {
    console.log('Validating navigation structure...');

    // Check if sidebar configuration includes module-4 items
    const sidebarPath = './sidebars.js';
    if (fs.existsSync(sidebarPath)) {
      const sidebarContent = fs.readFileSync(sidebarPath, 'utf8');

      this.results.navigation.push({
        file: sidebarPath,
        valid: sidebarContent.includes('module-4'),
        issues: sidebarContent.includes('module-4') ? [] : ['Module 4 not found in sidebar']
      });
    } else {
      this.results.navigation.push({
        file: sidebarPath,
        valid: false,
        issues: ['Sidebar configuration file not found']
      });
    }
  }

  async validateInternalLinks() {
    console.log('Validating internal links...');

    const mdxFiles = this.getMdxFiles();

    for (const file of mdxFiles) {
      const content = fs.readFileSync(file, 'utf8');
      const links = this.extractInternalLinks(content);

      const fileDir = path.dirname(file);
      const validationResults = [];

      for (const link of links) {
        const isValid = this.validateLink(link, fileDir);
        validationResults.push({ link, valid: isValid });
      }

      this.results.links.push({
        file: file,
        valid: validationResults.every(r => r.valid),
        issues: validationResults.filter(r => !r.valid).map(r => `Invalid link: ${r.link}`)
      });
    }
  }

  extractInternalLinks(content) {
    // Match markdown links [text](path) and relative paths
    const linkRegex = /\[([^\]]+)\]\(([^)]+)\)/g;
    const matches = [...content.matchAll(linkRegex)];
    return matches.map(match => match[2]).filter(link =>
      link.startsWith('./') || link.startsWith('../') || link.startsWith('/')
    );
  }

  validateLink(link, fromDir) {
    try {
      let targetPath;

      if (link.startsWith('/')) {
        // Absolute path from docs root
        targetPath = path.join(process.cwd(), 'docs', link.substring(1));
      } else {
        // Relative path
        targetPath = path.resolve(fromDir, link);
      }

      // Add .mdx extension if not present
      if (!targetPath.endsWith('.mdx') && !targetPath.endsWith('.md')) {
        targetPath += '.mdx';
      }

      return fs.existsSync(targetPath);
    } catch (error) {
      return false;
    }
  }

  async testBuild() {
    console.log('Testing Docusaurus build...');

    try {
      // Attempt to build the site
      const buildOutput = execSync('yarn build', {
        encoding: 'utf8',
        cwd: process.cwd(),
        timeout: 300000 // 5 minutes timeout
      });

      this.results.buildSuccess = true;
      console.log('Build completed successfully');

      // Verify build output contains expected content
      const buildPath = './build';
      if (fs.existsSync(buildPath)) {
        const htmlFiles = this.getHtmlFiles(buildPath);
        console.log(`Build successful with ${htmlFiles.length} HTML files generated`);
      }
    } catch (error) {
      this.results.buildSuccess = false;
      console.error('Build failed:', error.message);
      console.error('Build output:', error.stdout);
      console.error('Build errors:', error.stderr);
    }
  }

  getHtmlFiles(dir) {
    const files = [];
    const walk = (currentDir) => {
      const dirItems = fs.readdirSync(currentDir);
      for (const item of dirItems) {
        const fullPath = path.join(currentDir, item);
        const stat = fs.statSync(fullPath);

        if (stat.isDirectory()) {
          walk(fullPath);
        } else if (item.endsWith('.html')) {
          files.push(fullPath);
        }
      }
    };

    walk(dir);
    return files;
  }
}

// Run validation if called directly
if (require.main === module) {
  (async () => {
    const validator = new BuildValidator();
    const results = await validator.validateAll();

    console.log('\nVALIDATION RESULTS:');
    console.log('==================');

    // MDX Syntax Results
    console.log(`\nMDX Syntax Validation: ${results.mdxSyntax.filter(r => r.valid).length}/${results.mdxSyntax.length} files valid`);
    results.mdxSyntax.forEach(result => {
      if (!result.valid) {
        console.log(`  ❌ ${result.file}: ${result.issues.join(', ')}`);
      }
    });

    // Navigation Results
    console.log(`\nNavigation Validation: ${results.navigation.filter(r => r.valid).length}/${results.navigation.length} files valid`);
    results.navigation.forEach(result => {
      if (!result.valid) {
        console.log(`  ❌ ${result.file}: ${result.issues.join(', ')}`);
      }
    });

    // Links Results
    console.log(`\nLink Validation: ${results.links.filter(r => r.valid).length}/${results.links.length} files valid`);
    results.links.forEach(result => {
      if (!result.valid) {
        console.log(`  ❌ ${result.file}: ${result.issues.join(', ')}`);
      }
    });

    // Build Result
    console.log(`\nBuild Success: ${results.buildSuccess ? '✅ YES' : '❌ NO'}`);

    // Overall result
    const allValid = results.mdxSyntax.every(r => r.valid) &&
                     results.navigation.every(r => r.valid) &&
                     results.links.every(r => r.valid) &&
                     results.buildSuccess;

    console.log(`\nOVERALL RESULT: ${allValid ? '✅ ALL VALID' : '❌ ISSUES FOUND'}`);

    process.exit(allValid ? 0 : 1);
  })();
}

module.exports = BuildValidator;
```
