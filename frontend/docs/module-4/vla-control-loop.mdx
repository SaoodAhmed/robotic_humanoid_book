# Integrated VLA Control Loop

## Overview
This document describes the implementation of the integrated Vision-Language-Action (VLA) control loop. This system coordinates vision, language processing, and physical action execution in a continuous cycle, enabling the humanoid robot to execute complex tasks that require coordination between perception, language understanding, and physical actions.

## Architecture

The integrated VLA control loop consists of the following interconnected components:

1. **Perception System**: Processes visual and sensor data to understand the environment
2. **Language System**: Interprets voice commands and generates action plans
3. **Action Execution**: Coordinates ROS 2 action servers for navigation, manipulation, and perception
4. **State Management**: Maintains system state and handles transitions
5. **Error Recovery**: Manages failures and attempts recovery
6. **Logging & Monitoring**: Tracks state transitions and execution progress

class VLAControlLoop:
    def __init__(self, config):

        self.current_action_plan = None

    def start_control_loop(self):
        """
        Start the main VLA control loop
        """
        self.is_running = True

        # Start voice processing
        self.voice_processor.start_processing(self.process_command)

        self.control_thread = threading.Thread(target=self._control_loop, daemon=True)

        """

    def command_callback(self, msg: String):
        """
        ROS callback for command messages
        """
        command_text = msg.data
        rospy.loginfo(f"Received command: {command_text}")

        # Process the command through the control loop
        self.process_command(command_text, confidence=1.0)  # High confidence for direct commands

    def process_command(self, command_text: str, confidence: float):
        """
        Process a voice or direct command through the VLA pipeline
        """
        rospy.loginfo(f"Processing command: '{command_text}' with confidence {confidence}")

        # Validate confidence threshold
        if confidence < self.config.get('confidence_threshold', 0.5):
            rospy.logwarn(f"Command confidence {confidence} below threshold, ignoring")
            return

        # Transition to PROCESSING state
        if self.state_machine.handle_audio_capture_complete():
            rospy.loginfo("Transitioned to PROCESSING state")
        else:
            rospy.logerr("Failed to transition to PROCESSING state")
            return

        # Generate action plan using LLM
        rospy.loginfo("Generating action plan with LLM...")
        action_plan = self.language_translator.translate_command_to_action_plan(
            command_text,
            self.environment_context,
            self.robot_capabilities
        )

        if not action_plan:
            rospy.logerr("Failed to generate action plan")
            self.state_machine.handle_plan_generation_failure()
            return

        rospy.loginfo(f"Generated action plan with {len(action_plan)} actions")

        # Validate the action plan
        is_valid, validation_msg = self.language_translator.validate_action_plan(
            action_plan,
            self.robot_capabilities
        )

        if not is_valid:
            rospy.logerr(f"Action plan validation failed: {validation_msg}")
            self.state_machine.handle_plan_generation_failure()
            return

        # Apply safety guardrails
        filtered_plan = self.language_translator.apply_guardrails(action_plan)
        rospy.loginfo(f"Applied guardrails, plan now has {len(filtered_plan)} actions")

        # Transition to EXECUTING state
        if self.state_machine.handle_plan_generation_success():
            rospy.loginfo("Transitioned to EXECUTING state")

            # Execute the action plan
            self.execute_action_plan(filtered_plan)
        else:
            rospy.logerr("Failed to transition to EXECUTING state")
            self.state_machine.handle_error_occurred("Failed to transition to execution state")

    def execute_action_plan(self, action_plan: List[Dict]):
        """
        Execute a plan of actions in sequence
        """
        rospy.loginfo(f"Starting execution of action plan with {len(action_plan)} actions")

        self.current_action_plan = action_plan
        self.current_action_index = 0
        action_count = len(action_plan)

        # Update state machine to EXECUTING
        # (This should already be handled by the calling function)

        for i, action in enumerate(action_plan):
            rospy.loginfo(f"Executing action {i+1}/{action_count}: {action['type']}")

            # Update current action index
            self.current_action_index = i

            # Execute the action
            success = self.execute_single_action(action)

            if not success:
                rospy.logerr(f"Action {i+1} failed: {action['type']}")

                # Attempt recovery
                recovery_success = self.attempt_recovery(action, i, action_plan)

                if not recovery_success:
                    rospy.logerr("Recovery failed, terminating action plan")
                    self.state_machine.handle_action_execution_failure()
                    return

            # Publish execution progress
            self.publish_execution_feedback(i + 1, action_count, action['type'])

        # Plan completed successfully
        rospy.loginfo("Action plan completed successfully")
        self.state_machine.handle_action_execution_complete()

        # Clear current plan
        self.current_action_plan = None
        self.current_action_index = 0

    def execute_single_action(self, action: Dict) -> bool:
        """
        Execute a single action and return success status
        """
        action_type = action['type']
        parameters = action.get('parameters', {})

        rospy.loginfo(f"Executing action: {action_type} with params: {parameters}")

        try:
            if action_type == 'navigate_to_pose':
                return self.execute_navigation_action(parameters)
            elif action_type == 'detect_object':
                return self.execute_perception_action(parameters)
            elif action_type == 'grasp_object':
                return self.execute_manipulation_action(parameters)
            elif action_type == 'release_object':
                return self.execute_manipulation_action(parameters)
            elif action_type == 'speak':
                return self.execute_speech_action(parameters)
            elif action_type == 'wait':

            elif action_type == 'perceive_environment':

            else:
                rospy.logerr(f"Unknown action type: {action_type}")
                return False

        except Exception as e:
            rospy.logerr(f"Error executing action {action_type}: {e}")
            return False

    def attempt_recovery(self, failed_action: Dict, action_index: int, action_plan: List[Dict]) -> bool:
        """
        Attempt to recover from a failed action
        """
        rospy.loginfo(f"Attempting recovery from failed action: {failed_action['type']}")

        # Check if we've exceeded recovery attempts
        if not self.state_machine.can_attempt_recovery():
            rospy.logerr("Max recovery attempts reached")
            return False

        # Handle specific recovery strategies based on action type
        action_type = failed_action['type']

        if action_type == 'navigate_to_pose':
            return self.recover_navigation_failure(failed_action, action_index, action_plan)
        elif action_type == 'detect_object':
            return self.recover_perception_failure(failed_action, action_index, action_plan)
        elif action_type == 'grasp_object':
            return self.recover_manipulation_failure(failed_action, action_index, action_plan)
        else:
            # For other action types, try the same action again
            return self.retry_action(failed_action)

    def retry_action(self, action: Dict) -> bool:
        """
        Simple retry strategy for any action
        """

        try:

            if success:
                rospy.loginfo("Action retry successful")
                return True
            else:
                rospy.loginfo("Action retry failed")
                return False

    def publish_execution_feedback(self, current_step: int, total_steps: int, current_action: str):
        """
        Publish feedback about action execution progress
        """
        feedback_msg = {
            'current_action': current_action,

            'total_steps': total_steps,
            'progress_percentage': (current_step / total_steps) * 100.0,
            'state': self.state_machine.current_state.value
        }

        feedback_str = String()
        feedback_str.data = str(feedback_msg)
        self.state_pub.publish(feedback_str)

    def _control_loop(self):
        """
        Main control loop thread function
        """
        rate = rospy.Rate(10)  # 10 Hz control loop

        while self.is_running and not rospy.is_shutdown():
            try:
                # Publish current state
                state_info = self.state_machine.get_current_state_info()
                state_msg = String()
                state_msg.data = str(state_info)
                self.state_pub.publish(state_msg)

                # Additional control loop logic can go here

            except Exception as e:
                rospy.logerr(f"Error in control loop: {e}")

    def get_execution_status(self) -> Dict[str, Any]:

def handle_dynamic_obstacles(self, current_plan: List[Dict]) -> List[Dict]:
    """
    Detect and handle obstacles that appear during execution
    """
    # Check for new obstacles in navigation actions
    for i, action in enumerate(current_plan):
        if action['type'] == 'navigate_to_pose':
            # Check if path is still clear
            if not self.is_path_clear(action['parameters']):
                # Recalculate path avoiding new obstacles
                new_plan = self.recalculate_path(action['parameters'])
                if new_plan:
                    # Update the plan with the new route
                    current_plan[i] = new_plan[0]  # Assuming single action replacement
                else:
                    # Handle path recalculation failure
                    self.state_machine.handle_error_occurred("Path blocked, unable to recalculate")
                    return None

    return current_plan
